# 커스텀 애니메이션으로 전문가급 효과 만들기

이번 튜토리얼에서는 Flitter의 고급 애니메이션 기법을 배워서 완전히 커스터마이징된 애니메이션 효과를 만들어봅시다. 재사용 가능한 애니메이션 위젯, CustomPaint와의 결합, 물리 기반 애니메이션까지 다루겠습니다.

## 🎯 학습 목표

이 튜토리얼을 완료하면 다음을 할 수 있게 됩니다:

- [ ] AnimatedWidget을 상속해서 재사용 가능한 애니메이션 위젯 만들기
- [ ] CustomPaint와 애니메이션을 결합해서 그래픽 애니메이션 구현하기
- [ ] 복잡한 수학적 계산을 통한 물리 기반 애니메이션 만들기
- [ ] 여러 애니메이션을 조합한 복합 효과 구현하기
- [ ] 성능을 고려한 애니메이션 최적화 기법 적용하기
- [ ] 실제 앱에서 사용할 수 있는 고급 애니메이션 패턴 이해하기

## 🚀 커스텀 애니메이션의 필요성

지금까지 배운 AnimatedContainer, AnimatedOpacity 등은 미리 정의된 애니메이션입니다. 하지만 실제 앱 개발에서는 다음과 같은 커스텀 애니메이션이 필요합니다:

- **브랜드 특화 효과**: 회사만의 독특한 애니메이션
- **복잡한 그래픽**: 차트, 그래프, 로더 등
- **물리 시뮬레이션**: 중력, 탄성, 마찰 등
- **게임 효과**: 파티클, 폭발, 물결 등
- **재사용성**: 여러 곳에서 사용할 수 있는 컴포넌트

## 📋 단계별 실습

### 1단계: CustomPaint와 애니메이션 기본 결합

먼저 CustomPaint를 사용해서 그래픽 요소를 애니메이션해봅시다:

```javascript
import { CustomPaint, Canvas } from "@meursyphus/flitter";

// 커스텀 페인터 클래스
class CirclePainter {
  constructor(radius, color) {
    this.radius = radius;
    this.color = color;
  }

  paint(canvas, size) {
    const center = { x: size.width / 2, y: size.height / 2 };
    
    // 원 그리기
    canvas.beginPath();
    canvas.arc(center.x, center.y, this.radius, 0, 2 * Math.PI);
    canvas.fillStyle = this.color;
    canvas.fill();
    
    // 테두리
    canvas.beginPath();
    canvas.arc(center.x, center.y, this.radius, 0, 2 * Math.PI);
    canvas.strokeStyle = '#333';
    canvas.lineWidth = 2;
    canvas.stroke();
  }
}

class AnimatedCircle extends StatefulWidget {
  createState() {
    return new AnimatedCircleState();
  }
}

class AnimatedCircleState extends State<AnimatedCircle> {
  controller = null;
  radiusAnimation = null;
  colorAnimation = null;

  initState(context) {
    super.initState(context);
    
    this.controller = AnimationController({ duration: 2000 });
    
    // 반지름 애니메이션
    this.radiusAnimation = Tween({ begin: 20, end: 100 }).animated(
      CurvedAnimation({
        parent: this.controller,
        curve: "bounceOut"
      })
    );
    
    // 색상 변화를 위한 비율 애니메이션
    this.colorAnimation = Tween({ begin: 0, end: 1 }).animated(this.controller);
    
    this.controller.addListener(() => {
      this.setState();
    });
  }

  dispose() {
    this.controller.dispose();
    super.dispose();
  }

  getInterpolatedColor() {
    const t = this.colorAnimation.value;
    const r = Math.round(255 * (1 - t) + 100 * t);
    const g = Math.round(100 * (1 - t) + 255 * t);
    const b = 100;
    return `rgb(${r}, ${g}, ${b})`;
  }

  build(context) {
    return Column({
      children: [
        GestureDetector({
          onClick: () => {
            if (this.controller.isCompleted) {
              this.controller.reverse();
            } else {
              this.controller.forward();
            }
          },
          child: Container({
            width: 150,
            height: 50,
            color: '#6C5CE7',
            child: Text("원 애니메이션")
          })
        }),
        
        Container({
          width: 250,
          height: 250,
          child: CustomPaint({
            painter: new CirclePainter(
              this.radiusAnimation.value,
              this.getInterpolatedColor()
            )

        })
      ]
    });
  }
}
```

```javascript
class WavePainter {
  constructor(amplitude, frequency, phase, color) {
    this.amplitude = amplitude;
    this.frequency = frequency;
    this.phase = phase;
    this.color = color;
  }

  paint(canvas, size) {
    const centerY = size.height / 2;
    const step = 2;
    
    canvas.beginPath();
    canvas.strokeStyle = this.color;
    canvas.lineWidth = 3;
    
    for (let x = 0; x <= size.width; x += step) {
      const y = centerY + Math.sin((x * this.frequency + this.phase) * Math.PI / 180) * this.amplitude;
      
      if (x === 0) {
        canvas.moveTo(x, y);
      } else {
        canvas.lineTo(x, y);
      }
    }
    
    canvas.stroke();
  }
}
```

### 3단계: 파티클 시스템 애니메이션

여러 개의 파티클이 동시에 움직이는 복잡한 애니메이션을 만들어봅시다:

```javascript
class Particle {
  constructor(x, y, vx, vy, color, size) {
    this.x = x;
    this.y = y;
    this.vx = vx;  // x 방향 속도
    this.vy = vy;  // y 방향 속도
    this.color = color;
    this.size = size;
    this.life = 1.0;  // 생명력 (1.0 ~ 0.0)
  }

  update(deltaTime, width, height) {
    this.x += this.vx * deltaTime;
    this.y += this.vy * deltaTime;
    this.vy += 0.2 * deltaTime;  // 중력 효과
    this.life -= 0.01 * deltaTime;

    // 경계 반사
    if (this.x <= 0 || this.x >= width) {
      this.vx *= -0.8;
      this.x = Math.max(0, Math.min(width, this.x));
    }
    if (this.y >= height) {
      this.vy *= -0.7;
      this.y = height;
    }

    return this.life > 0;
  }
}

class ParticleSystemPainter {
  constructor(particles) {
    this.particles = particles;
  }

  paint(canvas, size) {
    this.particles.forEach(particle => {
      const alpha = particle.life;
      
      canvas.beginPath();
      canvas.arc(particle.x, particle.y, particle.size, 0, 2 * Math.PI);
      canvas.fillStyle = `rgba(255, 100, 100, ${alpha})`;
      canvas.fill();
    });
  }
}

class ParticleSystem extends StatefulWidget {
  createState() {
    return new ParticleSystemState();
  }
}

class ParticleSystemState extends State<ParticleSystem> {
  particles = [];
  animationId = null;
  lastTime = 0;

  initState(context) {
    super.initState(context);
  }

  dispose() {
    if (this.animationId) {
      cancelAnimationFrame(this.animationId);
    }
    super.dispose();
  }

  createParticles(x, y) {
    for (let i = 0; i < 20; i++) {
      const angle = (Math.PI * 2 * i) / 20;
      const speed = Math.random() * 5 + 2;
      const vx = Math.cos(angle) * speed;
      const vy = Math.sin(angle) * speed - 5;  // 위쪽으로 초기 속도
      
      this.particles.push(new Particle(
        x, y, vx, vy,
        `hsl(${Math.random() * 360}, 70%, 60%)`,
        Math.random() * 5 + 2
      ));
    }
  }

  startAnimation() {
    const animate = (currentTime) => {
      const deltaTime = this.lastTime ? (currentTime - this.lastTime) / 16 : 1;
      this.lastTime = currentTime;

      // 파티클 업데이트
      this.particles = this.particles.filter(particle => 
        particle.update(deltaTime, 300, 200)
      );

      this.setState(() => {});
      this.animationId = requestAnimationFrame(animate);
    };

    this.animationId = requestAnimationFrame(animate);
  }

  explode() {
    this.createParticles(150, 100);  // 중앙에서 폭발
    if (!this.animationId) {
      this.startAnimation();
    }
  }

  build(context) {
    return Column({
      children: [
        GestureDetector({
          onClick: () => this.explode(),
          child: Container({
            width: 120,
            height: 50,
            color: '#FF5722',
            child: Text("폭발 효과!")
          })
        }),
        
        Container({
          width: 300,
          height: 200,
          color: '#000',
          child: CustomPaint({
            painter: new ParticleSystemPainter(this.particles)
          })
        }),
        
        Text(`활성 파티클: ${this.particles.length}`)
      ]
    });
  }
}
```

```javascript
class LoadingSpinnerPainter {
  constructor(progress, dotCount, dotSize, color) {
    this.progress = progress;
    this.dotCount = dotCount;
    this.dotSize = dotSize;
    this.color = color;
  }

  paint(canvas, size) {
    const center = { x: size.width / 2, y: size.height / 2 };
    const radius = Math.min(size.width, size.height) / 2 - this.dotSize;

    for (let i = 0; i < this.dotCount; i++) {
      const angle = (i / this.dotCount) * 2 * Math.PI;
      const dotProgress = (this.progress * this.dotCount - i) % this.dotCount;
      const alpha = Math.max(0, Math.min(1, dotProgress));
      
      const x = center.x + Math.cos(angle) * radius;
      const y = center.y + Math.sin(angle) * radius;
      
      canvas.beginPath();
      canvas.arc(x, y, this.dotSize, 0, 2 * Math.PI);
      canvas.fillStyle = `rgba(33, 150, 243, ${alpha})`;
      canvas.fill();
    }
  }
}
```

```javascript
class PulsingButton extends StatefulWidget {
  constructor(props) {
    super();
    this.child = props.child;
    this.onTap = props.onTap;
    this.color = props.color || '#2196F3';

    return new PulsingButtonState();

}

class PulsingButtonState extends State<PulsingButton> {
  controller = null;
  scaleAnimation = null;
  opacityAnimation = null;

  initState(context) {
    super.initState(context);
    
    this.controller = AnimationController({ duration: this.widget.duration });
    
    this.scaleAnimation = Tween({ begin: 1.0, end: 1.2 }).animated(
      CurvedAnimation({
        parent: this.controller,
        curve: "easeInOut"
      })
    );
    
    this.opacityAnimation = Tween({ begin: 1.0, end: 0.7 }).animated(
      CurvedAnimation({
        parent: this.controller,
        curve: "easeInOut"
      })
    );
    
    this.controller.addListener(() => {
      this.setState();
    });
    
    // 자동으로 펄싱 시작
    this.controller.repeat({ reverse: true });
  }

  dispose() {
    this.controller.dispose();
    super.dispose();
  }

    return GestureDetector({
      onClick: this.widget.onTap,
      child: Transform.scale({
        scale: this.scaleAnimation.value,
        child: AnimatedOpacity({
          duration: 0,  // Controller가 제어
          opacity: this.opacityAnimation.value,
          child: Container({
            color: this.widget.color,
            child: this.widget.child
          })
        })
      })
    });
  }
}
```

### 6단계: 물리 기반 스프링 애니메이션

실제 물리 법칙을 시뮬레이션하는 스프링 애니메이션을 구현해봅시다:

```javascript
class SpringPhysics {
  constructor(mass = 1, stiffness = 100, damping = 10) {
    this.mass = mass;
    this.stiffness = stiffness;
    this.damping = damping;
    
    this.position = 0;
    this.velocity = 0;
    this.target = 0;
  }

  update(deltaTime) {
    const force = -this.stiffness * (this.position - this.target);
    const dampingForce = -this.damping * this.velocity;
    const acceleration = (force + dampingForce) / this.mass;
    
    this.velocity += acceleration * deltaTime;
    this.position += this.velocity * deltaTime;
    
    return Math.abs(this.velocity) > 0.1 || Math.abs(this.position - this.target) > 0.1;
  }

  setTarget(target) {
    this.target = target;
  }
}

class SpringAnimation extends StatefulWidget {
  createState() {
    return new SpringAnimationState();
  }
}

class SpringAnimationState extends State<SpringAnimation> {
  spring = null;
  animationId = null;
  lastTime = 0;

  initState(context) {
    super.initState(context);
    this.spring = new SpringPhysics(1, 200, 20);
  }

  dispose() {
    if (this.animationId) {
      cancelAnimationFrame(this.animationId);
    }
    super.dispose();
  }

  startSpringAnimation(target) {
    this.spring.setTarget(target);
    
    const animate = (currentTime) => {
      const deltaTime = this.lastTime ? (currentTime - this.lastTime) / 1000 : 0.016;
      this.lastTime = currentTime;

      const isActive = this.spring.update(deltaTime);
      this.setState(() => {});

      if (isActive) {
        this.animationId = requestAnimationFrame(animate);
      } else {
        this.animationId = null;
      }
    };

    if (!this.animationId) {
      this.animationId = requestAnimationFrame(animate);
```

```javascript
  paint(canvas, size) {
    const centerY = size.height / 2;
    const step = 2;
    
    canvas.clearRect(0, 0, size.width, size.height);
    
    // 격자 그리기
    canvas.strokeStyle = '#e0e0e0';
    canvas.lineWidth = 1;
    
    for (let x = 0; x < size.width; x += 20) {
      canvas.beginPath();
      canvas.moveTo(x, 0);
      canvas.lineTo(x, size.height);
      canvas.stroke();
    }
    
    for (let y = 0; y < size.height; y += 20) {
      canvas.beginPath();
      canvas.moveTo(0, y);
      canvas.lineTo(size.width, y);
      canvas.stroke();
    }
    
    // 심박 파형 그리기
    canvas.strokeStyle = '#e74c3c';
    canvas.lineWidth = 2;
    canvas.beginPath();
    
    for (let x = 0; x < size.width; x += step) {
      const timePoint = (this.currentTime - (size.width - x) * 10) / 1000;
      let y = centerY;
      
      // 각 박동에 대해 파형 계산
      this.beats.forEach(beatTime => {
        const timeDiff = timePoint - beatTime;
        if (timeDiff >= 0 && timeDiff <= 0.3) {
          // QRS 복합파 시뮬레이션
          if (timeDiff < 0.05) {
            y += Math.sin(timeDiff * Math.PI / 0.05) * 30;
          } else if (timeDiff < 0.1) {
            y -= Math.sin((timeDiff - 0.05) * Math.PI / 0.05) * 60;
          } else if (timeDiff < 0.15) {
            y += Math.sin((timeDiff - 0.1) * Math.PI / 0.05) * 40;
          } else if (timeDiff < 0.25) {
            y += Math.sin((timeDiff - 0.15) * Math.PI / 0.1) * 15;
          }
        }
      });
      
      if (x === 0) {
        canvas.moveTo(x, y);
      } else {
        canvas.lineTo(x, y);
      }
    }
    
    canvas.stroke();
  }
```

```javascript
class Ball {
  constructor(x, y, radius, color) {
    this.x = x;
    this.y = y;
    this.vx = 0;
    this.vy = 0;
    this.radius = radius;
    this.color = color;
    this.isDragging = false;
    this.mass = radius * 0.1;
  }

  update(deltaTime, width, height, gravity = 9.8) {
    if (!this.isDragging) {
      this.vy += gravity * deltaTime;
      this.x += this.vx * deltaTime;
      this.y += this.vy * deltaTime;

      // 경계 충돌
      if (this.x - this.radius <= 0 || this.x + this.radius >= width) {
        this.vx *= -0.8;
        this.x = Math.max(this.radius, Math.min(width - this.radius, this.x));
      }
      
      if (this.y + this.radius >= height) {
        this.vy *= -0.8;
        this.y = height - this.radius;
        this.vx *= 0.95;  // 마찰
      }
      
      if (this.y - this.radius <= 0) {
        this.vy *= -0.8;
        this.y = this.radius;
      }
    }
  }

  contains(px, py) {
    const dx = px - this.x;
    const dy = py - this.y;
    return dx * dx + dy * dy <= this.radius * this.radius;
  }
}

class PhysicsSimulationState extends State<PhysicsSimulation> {
  balls = [];
  animationId = null;

    super.initState(context);

      new Ball(100, 50, 20, '#e74c3c'),

    this.startAnimation();

  dispose() {
    if (this.animationId) {
      cancelAnimationFrame(this.animationId);
    }
    super.dispose();
  }

  startAnimation() {
    const animate = (currentTime) => {
      const deltaTime = this.lastTime ? (currentTime - this.lastTime) / 1000 : 0.016;
      this.lastTime = currentTime;

      this.balls.forEach(ball => ball.update(deltaTime, 300, 200));
      
      this.setState(() => {});
      this.animationId = requestAnimationFrame(animate);
    };
    
    this.animationId = requestAnimationFrame(animate);
  }

  handleMouseDown(event) {
    const rect = event.target.getBoundingClientRect();
    const x = event.clientX - rect.left;
    const y = event.clientY - rect.top;
    
    this.draggingBall = this.balls.find(ball => ball.contains(x, y));
    if (this.draggingBall) {
      this.draggingBall.isDragging = true;
      this.lastMousePos = { x, y };

    if (this.draggingBall && this.lastMousePos) {
      const rect = event.target.getBoundingClientRect();

      this.draggingBall.y = y;
      
      // 속도 계산 (드래그 릴리즈 시 사용)
      this.draggingBall.vx = (x - this.lastMousePos.x) * 10;
      this.draggingBall.vy = (y - this.lastMousePos.y) * 10;
      
      this.lastMousePos = { x, y };
    }
  }

    if (this.draggingBall) {
      this.draggingBall.isDragging = false;
      this.draggingBall = null;
      this.lastMousePos = null;

class BallsPainter {
  constructor(balls) {
    this.balls = balls;
  }

  paint(canvas, size) {
    // 배경
    canvas.fillStyle = '#ffffff';
    canvas.fillRect(0, 0, size.width, size.height);
    
    // 각 공 그리기
    this.balls.forEach(ball => {
      canvas.beginPath();
      canvas.arc(ball.x, ball.y, ball.radius, 0, 2 * Math.PI);
      canvas.fillStyle = ball.color;
      canvas.fill();
      
      canvas.beginPath();
      canvas.arc(ball.x, ball.y, ball.radius, 0, 2 * Math.PI);
      canvas.strokeStyle = '#333';
      canvas.lineWidth = 2;
      canvas.stroke();
      
      // 드래그 중인 공 강조
      if (ball.isDragging) {
        canvas.beginPath();
        canvas.arc(ball.x, ball.y, ball.radius + 5, 0, 2 * Math.PI);
        canvas.strokeStyle = '#ff0000';
        canvas.lineWidth = 3;
        canvas.stroke();
      }
    });
  }
```
