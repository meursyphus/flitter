## 🎪 Draggable과 DragTarget 이해하기

### 🔄 핵심 개념

**Draggable**: 사용자가 드래그할 수 있는 위젯
**DragTarget**: 드래그된 위젯을 받을 수 있는 드롭 영역

```typescript
// 기본 구조
Draggable({
  data: myData,          // 전달할 데이터
  child: normalWidget,   // 평상시 표시되는 위젯
  feedback: dragWidget,  // 드래그 중 표시되는 위젯
  onDragStart: () => {}, // 드래그 시작 시
  onDragEnd: () => {}    // 드래그 종료 시
})

DragTarget({
  onAccept: (data) => {},     // 드롭 성공 시
  onDragEnter: () => {},      // 드래그가 영역에 들어올 때
  onDragLeave: () => {},      // 드래그가 영역을 벗어날 때
  builder: (context, candidateData, rejectedData) => {
    return yourWidget; // 드롭 영역 UI
  }
})
```

class ConditionalDragTarget extends StatefulWidget {
  constructor(private props: {
    acceptedType: string;
    onDrop: (data: any) => void;
  }) {
    super();
  }
  
  createState() {
    return new ConditionalDragTargetState();
  }
}

class ConditionalDragTargetState extends State {
  isHovering = false;
  isValidTarget = false;
  
  build(context) {
    const { acceptedType, onDrop } = this.widget.props;
    
    return DragTarget({
      // 특정 타입만 받아들이기
      onWillAccept: (data) => {
        const isValid = data && data.type === acceptedType;
        this.setState(() => {
          this.isValidTarget = isValid;
        });
        return isValid;
      },
      
      onAccept: (data) => {
        onDrop(data);
        this.setState(() => {
          this.isHovering = false;
          this.isValidTarget = false;
        });
      },
      
      onDragEnter: (data) => {
        this.setState(() => {
          this.isHovering = true;
        });
      },
      
      onDragLeave: (data) => {
        this.setState(() => {
          this.isHovering = false;
          this.isValidTarget = false;
        });
      },

class ScrollableDragArea extends StatefulWidget {
  createState() {
    return new ScrollableDragAreaState();
  }
}

class ScrollableDragAreaState extends State {
  scrollOffset = 0;
  isAutoScrolling = false;
  autoScrollTimer = null;
  
  handleDragUpdate(details) {
    const scrollThreshold = 50; // 가장자리에서 스크롤 시작하는 거리
    const scrollSpeed = 5;
    
    // 상단 가장자리 근처
    if (details.globalPosition.y < scrollThreshold) {
      this.startAutoScroll(-scrollSpeed);
    }
    // 하단 가장자리 근처
    else if (details.globalPosition.y > window.innerHeight - scrollThreshold) {
      this.startAutoScroll(scrollSpeed);
    }
    // 스크롤 중지
    else {
      this.stopAutoScroll();
    }
  }
  
  startAutoScroll(speed) {
    if (this.isAutoScrolling) return;
    
    this.isAutoScrolling = true;
    this.autoScrollTimer = setInterval(() => {
      this.setState(() => {
        this.scrollOffset = Math.max(0, this.scrollOffset + speed);
      });
    }, 16); // 60fps
  }
  
  stopAutoScroll() {
    if (this.autoScrollTimer) {
      clearInterval(this.autoScrollTimer);
      this.autoScrollTimer = null;
      this.isAutoScrolling = false;
    }
  }
  
  dispose() {
    this.stopAutoScroll();
    super.dispose();
  }
}
