import { Widget } from '@meursyphus/flitter-svelte';
import { Container, Column, Row, Text, TextStyle, Center, EdgeInsets, GestureDetector, AnimatedContainer, CustomPaint, Stack, Positioned } from '@meursyphus/flitter';

# 첫 번째 위젯 만들기

이 튜토리얼에서는 Flitter를 사용하여 인터랙티브한 막대 차트 위젯을 단계별로 만들어봅니다. 기본적인 정적 차트부터 시작해서 애니메이션과 인터랙션을 추가하는 과정을 배워봅시다.

## 시작하기 전에

Flitter가 설치되어 있고 개발 환경이 준비되어 있다고 가정합니다. 아직 설치하지 않았다면 [설치 가이드](/ko/installation)를 참고하세요.

## 1단계: 기본 막대 차트

먼저 가장 간단한 형태의 막대 차트를 만들어봅시다.

<Widget name="first-widget-step1">

    return Container({
      width: 400,

  })()}

### 코드 설명

1. **데이터 준비**: 간단한 레이블과 값을 가진 배열
2. **Container**: 전체 차트를 감싸는 컨테이너

## 2단계: 스타일 개선

## 3단계: 인터랙션 추가

사용자가 막대를 클릭하면 상세 정보를 보여주도록 만들어봅시다.

<Widget name="first-widget-step3">
  {(() => {
    const [selectedIndex, setSelectedIndex] = React.useState(null);
    
    const data = [
      { label: '1월', value: 30, color: '#ff6b6b', detail: '신년 프로모션' },
      { label: '2월', value: 50, color: '#4ecdc4', detail: '발렌타인 특수' },
      { label: '3월', value: 20, color: '#45b7d1', detail: '비수기' },
      { label: '4월', value: 40, color: '#f9ca24', detail: '봄 세일' }
    ];
    
    const maxValue = Math.max(...data.map(d => d.value));
    
    return Container({
      width: 450,
      height: 350,
      padding: EdgeInsets.all(20),
      color: 'white',
      child: Column({
        children: [
          Row({
            mainAxisAlignment: 'spaceBetween',
            children: [
              Column({
                crossAxisAlignment: 'start',
                children: [
                  Text('월별 판매량', {
                    style: TextStyle({
                      fontSize: 20,
                      fontWeight: 'bold',
                      color: '#212529'
                    })
                  }),
                  Text('클릭하여 상세 정보 확인', {
                    style: TextStyle({
                      fontSize: 12,
                      color: '#868e96'
                    })
                  })
                ]
              }),
              if (selectedIndex !== null) Container({
                padding: EdgeInsets.symmetric({ horizontal: 12, vertical: 6 }),
                decoration: {
                  color: data[selectedIndex].color + '20',
                  borderRadius: BorderRadius.circular(16),
                  border: {
                    color: data[selectedIndex].color,
                    width: 1
                  }
                },
                child: Text(data[selectedIndex].detail, {
                  style: TextStyle({
                    fontSize: 12,
                    color: data[selectedIndex].color,
                    fontWeight: 'bold'
                  })
                })
              })
            ]
          }),
          SizedBox({ height: 30 }),
          Expanded({
            child: Row({
              mainAxisAlignment: 'spaceEvenly',
              crossAxisAlignment: 'end',
              children: data.map((item, index) => 
                GestureDetector({
                  onTap: () => setSelectedIndex(
                    selectedIndex === index ? null : index
                  ),
                  child: Column({
                    mainAxisAlignment: 'end',
                    children: [
                      AnimatedContainer({
                        duration: 200,
                        padding: EdgeInsets.symmetric({ 
                          horizontal: 8, 
                          vertical: 4 
                        }),
                        decoration: selectedIndex === index ? {
                          color: item.color,
                          borderRadius: BorderRadius.circular(4)
                        } : null,
                        child: Text(`${item.value}`, {
                          style: TextStyle({
                            fontSize: 14,
                            fontWeight: 'bold',
                            color: selectedIndex === index ? 'white' : item.color
                          })
                        })
                      }),
                      SizedBox({ height: 4 }),
                      AnimatedContainer({
                        duration: 200,
                        width: selectedIndex === index ? 70 : 60,
                        height: (item.value / maxValue) * 150,
                        decoration: {
                          color: item.color,
                          borderRadius: BorderRadius.only({
                            topLeft: 8,
                            topRight: 8
                          }),
                          opacity: selectedIndex !== null && selectedIndex !== index ? 0.5 : 1
                        }
                      }),
                      SizedBox({ height: 8 }),
                      Text(item.label, {
                        style: TextStyle({
                          fontSize: 12,
                          fontWeight: selectedIndex === index ? 'bold' : 'normal',
                          color: selectedIndex === index ? item.color : '#495057'
                        })
                      })
                    ]
                  })
                })
              )
            })
          })
        ]
      })
    });
  })()}
</Widget>

### 인터랙션 기능

1. **GestureDetector**: 클릭 이벤트 처리
2. **상태 관리**: selectedIndex로 선택 상태 추적
3. **조건부 스타일**: 선택된 막대 강조
4. **AnimatedContainer**: 부드러운 전환 효과
5. **상세 정보**: 선택 시 추가 정보 표시

## 4단계: 애니메이션 추가

차트가 처음 나타날 때 애니메이션을 추가해봅시다.

<Widget name="first-widget-step4">
  {(() => {
    const [animationProgress, setAnimationProgress] = React.useState(0);
    const [selectedIndex, setSelectedIndex] = React.useState(null);
    
    React.useEffect(() => {
      const startTime = Date.now();
      const duration = 1000;
      
      const animate = () => {
        const elapsed = Date.now() - startTime;
        const progress = Math.min(elapsed / duration, 1);
        
        // Ease-out 애니메이션
        const eased = 1 - Math.pow(1 - progress, 3);
        setAnimationProgress(eased);
        
        if (progress < 1) {
          requestAnimationFrame(animate);
        }
      };
      
      requestAnimationFrame(animate);
    }, []);
    
    const data = [
      { label: '1월', value: 30, color: '#ff6b6b' },
      { label: '2월', value: 50, color: '#4ecdc4' },
      { label: '3월', value: 20, color: '#45b7d1' },
      { label: '4월', value: 40, color: '#f9ca24' }
    ];
    
    const maxValue = Math.max(...data.map(d => d.value));
    
    return Container({
      width: 450,
      height: 350,
      padding: EdgeInsets.all(20),
      color: 'white',
      child: Column({
        children: [
          Text('애니메이션 차트', {
            style: TextStyle({
              fontSize: 20,
              fontWeight: 'bold',
              color: '#212529'
            })
          }),
          SizedBox({ height: 30 }),
          Expanded({
            child: Row({
              mainAxisAlignment: 'spaceEvenly',
              crossAxisAlignment: 'end',
              children: data.map((item, index) => {
                const delay = index * 0.1;
                const itemProgress = Math.max(0, Math.min(1, (animationProgress - delay) / (1 - delay)));
                
                return GestureDetector({
                  onTap: () => setSelectedIndex(
                    selectedIndex === index ? null : index
                  ),
                  child: Column({
                    mainAxisAlignment: 'end',
                    children: [
                      AnimatedOpacity({
                        opacity: itemProgress,
                        duration: 300,
                        child: Text(`${item.value}`, {
                          style: TextStyle({
                            fontSize: 14,
                            fontWeight: 'bold',
                            color: item.color
                          })
                        })
                      }),
                      SizedBox({ height: 4 }),
                      Container({
                        width: 60,
                        height: (item.value / maxValue) * 150 * itemProgress,
                        decoration: {
                          color: item.color,
                          borderRadius: BorderRadius.only({
                            topLeft: 8,
                            topRight: 8
                          })
                        }
                      }),
                      SizedBox({ height: 8 }),
                      AnimatedOpacity({
                        opacity: itemProgress,
                        duration: 300,
                        child: Text(item.label, {
                          style: TextStyle({
                            fontSize: 12,
                            color: '#495057'
                          })
                        })
                      })
                    ]
                  })
                });
              })
            })
          })
        ]
      })
    });
  })()}
</Widget>

### 애니메이션 특징

1. **진입 애니메이션**: 막대가 아래에서 위로 성장
2. **순차적 지연**: 각 막대가 순서대로 나타남
3. **Ease-out 곡선**: 자연스러운 감속
4. **투명도 전환**: 텍스트가 페이드인

## 5단계: CustomPaint로 고급 차트

더 복잡한 시각화를 위해 CustomPaint를 사용해봅시다.

<Widget name="first-widget-step5">
  {(() => {
    const [hoveredIndex, setHoveredIndex] = React.useState(null);
    
    const data = [
      { label: 'Q1', value: 65, target: 70 },
      { label: 'Q2', value: 78, target: 75 },
      { label: 'Q3', value: 82, target: 80 },
      { label: 'Q4', value: 91, target: 85 }
    ];
    
    const AdvancedChartPainter = {
      paint: (canvas, size) => {
        const padding = 30;
        const chartWidth = size.width - padding * 2;
        const chartHeight = size.height - padding * 2;
        const barWidth = chartWidth / (data.length * 2);
        
        // 배경 그리드
        const gridPaint = Paint();
        gridPaint.color = '#f1f3f5';
        gridPaint.strokeWidth = 1;
        
        for (let i = 0; i <= 5; i++) {
          const y = padding + (chartHeight / 5) * i;
          canvas.drawLine(
            { x: padding, y },
            { x: size.width - padding, y },
            gridPaint
          );
          
          // Y축 레이블
          const textPaint = Paint();
          textPaint.color = '#868e96';
          textPaint.fontSize = 10;
          canvas.drawText(
            `${100 - i * 20}`,
            { x: 5, y: y - 5 },
            textPaint
          );
        }
        
        // 막대 그리기
        data.forEach((item, index) => {
          const x = padding + index * barWidth * 2 + barWidth / 2;
          const barHeight = (item.value / 100) * chartHeight;
          const targetY = padding + (1 - item.target / 100) * chartHeight;
          
          // 실제값 막대
          const barPaint = Paint();
          barPaint.color = hoveredIndex === index ? '#4c6ef5' : '#5c7cfa';
          barPaint.style = 'fill';
          
          canvas.drawRRect(
            RRect.fromRectAndCorners(
              Rect.fromLTWH(
                x,
                padding + chartHeight - barHeight,
                barWidth,
                barHeight
              ),
              topLeft: Radius.circular(4),
              topRight: Radius.circular(4)
            ),
            barPaint
          );
          
          // 목표선
          const targetPaint = Paint();
          targetPaint.color = '#ff6b6b';
          targetPaint.strokeWidth = 2;
          targetPaint.style = 'stroke';
          targetPaint.dashArray = [5, 3];
          
          canvas.drawLine(
            { x: x - 5, y: targetY },
            { x: x + barWidth + 5, y: targetY },
            targetPaint
          );
          
          // 레이블
          const labelPaint = Paint();
          labelPaint.color = '#495057';
          labelPaint.fontSize = 12;
          canvas.drawText(
            item.label,
            { x: x + barWidth / 2 - 10, y: size.height - 10 },
            labelPaint
          );
          
          // 호버 시 값 표시
          if (hoveredIndex === index) {
            const valuePaint = Paint();
            valuePaint.color = '#212529';
            valuePaint.fontSize = 14;
            valuePaint.fontWeight = 'bold';
            
            canvas.drawText(
              `${item.value}%`,
              { x: x + barWidth / 2 - 15, y: padding + chartHeight - barHeight - 20 },
              valuePaint
            );
          }
        });
      }
    };
    
    return Container({
      width: 450,
      height: 300,
      padding: EdgeInsets.all(20),
      color: 'white',
      child: Column({
        children: [
          Row({
            mainAxisAlignment: 'spaceBetween',
            children: [
              Text('분기별 목표 대비 실적', {
                style: TextStyle({
                  fontSize: 18,
                  fontWeight: 'bold',
                  color: '#212529'
                })
              }),
              Row({
                children: [
                  Container({
                    width: 20,
                    height: 3,
                    color: '#5c7cfa'
                  }),
                  SizedBox({ width: 4 }),
                  Text('실적', {
                    style: TextStyle({ fontSize: 12, color: '#495057' })
                  }),
                  SizedBox({ width: 12 }),
                  Container({
                    width: 20,
                    height: 3,
                    decoration: {
                      border: {
                        color: '#ff6b6b',
                        width: 2,
                        style: 'dashed'
                      }
                    }
                  }),
                  SizedBox({ width: 4 }),
                  Text('목표', {
                    style: TextStyle({ fontSize: 12, color: '#495057' })
                  })
                ]
              })
            ]
          }),
          SizedBox({ height: 20 }),
          Expanded({
            child: GestureDetector({
              onPointerMove: (e) => {
                const rect = e.currentTarget.getBoundingClientRect();
                const x = e.clientX - rect.left;
                const padding = 30;
                const barWidth = (rect.width - padding * 2) / 8;
                
                const index = Math.floor((x - padding) / (barWidth * 2));
                if (index >= 0 && index < data.length) {
                  setHoveredIndex(index);
                } else {
                  setHoveredIndex(null);
                }
              },
              onPointerLeave: () => setHoveredIndex(null),
              child: CustomPaint({
                painter: AdvancedChartPainter
              })
            })
          })
        ]
      })
    });
  })()}
</Widget>

### CustomPaint 장점

1. **완전한 제어**: 픽셀 단위로 그리기 가능
2. **성능**: 복잡한 그래픽도 효율적으로 렌더링
3. **커스터마이징**: 독특한 시각화 구현 가능
4. **정밀도**: 정확한 위치와 크기 계산
