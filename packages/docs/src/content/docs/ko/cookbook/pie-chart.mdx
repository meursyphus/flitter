import { Container, Stack, CustomPaint, GestureDetector, Text, TextStyle, Center, EdgeInsets, Column, Row, AnimatedContainer } from '@meursyphus/flitter';

# 기본 파이 차트

파이 차트는 전체에서 각 부분이 차지하는 비율을 시각적으로 표현하는 가장 효과적인 방법입니다. CustomPaint를 사용하여 인터랙티브한 파이 차트를 구현해봅시다.

## 간단한 파이 차트

기본적인 파이 차트 구현입니다.

<Widget name="cookbook-simple-pie-chart">
  {(() => {
    const data = [
      { label: 'Android', value: 45, color: '#51cf66' },
      { label: 'iOS', value: 30, color: '#4c6ef5' },
      { label: 'Web', value: 20, color: '#ff922b' },
      { label: '기타', value: 5, color: '#868e96' }
    ];
    
    const total = data.reduce((sum, item) => sum + item.value, 0);
    
    const PieChartPainter = {
      paint: (canvas, size) => {
        const centerX = size.width / 2;
        const centerY = size.height / 2;
        const radius = Math.min(centerX, centerY) - 20;
        
        let startAngle = -Math.PI / 2; // 12시 방향부터 시작
        
        data.forEach(item => {
          const sweepAngle = (item.value / total) * Math.PI * 2;
          const paint = Paint();
          paint.color = item.color;
          paint.style = 'fill';
          
          // 파이 조각 그리기
          const path = Path();
          path.moveTo(centerX, centerY);
          path.arcTo(
            Rect.fromCircle({ x: centerX, y: centerY }, radius),
            startAngle,
            sweepAngle,
            false
          );
          path.close();
          
          canvas.drawPath(path, paint);
          
          // 레이블 위치 계산
          const labelAngle = startAngle + sweepAngle / 2;
          const labelRadius = radius * 0.7;
          const labelX = centerX + labelRadius * Math.cos(labelAngle);
          const labelY = centerY + labelRadius * Math.sin(labelAngle);
          
          // 백분율 텍스트
          const textPaint = Paint();
          textPaint.color = 'white';
          textPaint.fontSize = 14;
          textPaint.fontWeight = 'bold';
          
          const percentage = Math.round((item.value / total) * 100);
          canvas.drawText(
            `${percentage}%`,
            { x: labelX - 15, y: labelY },
            textPaint
          );
          
          startAngle += sweepAngle;
        });
      }
    };
    
    return Container({
      width: 400,
      height: 350,
      padding: EdgeInsets.all(20),
      color: '#f8f9fa',
      child: Column({
        children: [
          Text('플랫폼별 사용자 분포', {
            style: TextStyle({
              fontSize: 18,
              fontWeight: 'bold',
              color: '#212529'
            })
          }),
          SizedBox({ height: 20 }),
          Container({
            height: 200,
            child: CustomPaint({
              painter: PieChartPainter
            })
          }),
          SizedBox({ height: 20 }),
          Row({
            mainAxisAlignment: 'center',
            children: data.map(item => 
              Container({
                margin: EdgeInsets.symmetric({ horizontal: 8 }),
                child: Row({
                  children: [
                    Container({
                      width: 12,
                      height: 12,
                      decoration: {
                        color: item.color,
                        borderRadius: BorderRadius.circular(2)
                      }
                    }),
                    SizedBox({ width: 4 }),
                    Text(item.label, {
                      style: TextStyle({
                        fontSize: 12,
                        color: '#495057'
                      })
                    })
                  ]
                })
              })
            )
          })
        ]
      })
    });
  })()}
</Widget>

## 인터랙티브 파이 차트

호버와 클릭으로 상호작용하는 파이 차트입니다.

<Widget name="cookbook-interactive-pie-chart">
  {(() => {
    const [selectedIndex, setSelectedIndex] = React.useState(null);
    const [hoveredIndex, setHoveredIndex] = React.useState(null);
    
    const data = [
      { label: '식비', value: 350000, color: '#ff6b6b' },
      { label: '교통비', value: 150000, color: '#4ecdc4' },
      { label: '주거비', value: 500000, color: '#45b7d1' },
      { label: '통신비', value: 80000, color: '#f9ca24' },
      { label: '여가비', value: 200000, color: '#6c5ce7' },
      { label: '기타', value: 120000, color: '#a29bfe' }
    ];
    
    const total = data.reduce((sum, item) => sum + item.value, 0);
    
    const InteractivePieChartPainter = {
      paint: (canvas, size) => {
        const centerX = size.width / 2;
        const centerY = size.height / 2;
        const radius = Math.min(centerX, centerY) - 40;
        
        let startAngle = -Math.PI / 2;
        
        data.forEach((item, index) => {
          const sweepAngle = (item.value / total) * Math.PI * 2;
          const isSelected = selectedIndex === index;
          const isHovered = hoveredIndex === index;
          
          // 선택/호버 시 확대
          const currentRadius = radius + (isSelected ? 15 : (isHovered ? 8 : 0));
          
          const paint = Paint();
          paint.color = item.color;
          paint.style = 'fill';
          
          // 그림자 효과
          if (isSelected || isHovered) {
            const shadowPaint = Paint();
            shadowPaint.color = 'rgba(0,0,0,0.2)';
            shadowPaint.style = 'fill';
            
            const shadowPath = Path();
            shadowPath.moveTo(centerX + 2, centerY + 2);
            shadowPath.arcTo(
              Rect.fromCircle({ x: centerX + 2, y: centerY + 2 }, currentRadius),
              startAngle,
              sweepAngle,
              false
            );
            shadowPath.close();
            canvas.drawPath(shadowPath, shadowPaint);
          }
          
          // 파이 조각
          const path = Path();
          path.moveTo(centerX, centerY);
          path.arcTo(
            Rect.fromCircle({ x: centerX, y: centerY }, currentRadius),
            startAngle,
            sweepAngle,
            false
          );
          path.close();
          
          canvas.drawPath(path, paint);
          
          // 테두리
          if (isSelected) {
            const borderPaint = Paint();
            borderPaint.color = 'white';
            borderPaint.style = 'stroke';
            borderPaint.strokeWidth = 3;
            canvas.drawPath(path, borderPaint);
          }
          
          startAngle += sweepAngle;
        });
        
        // 중앙 정보 표시
        if (selectedIndex !== null) {
          const selected = data[selectedIndex];
          const percentage = ((selected.value / total) * 100).toFixed(1);
          
          // 배경 원
          const bgPaint = Paint();
          bgPaint.color = 'white';
          bgPaint.style = 'fill';
          canvas.drawCircle({ x: centerX, y: centerY }, 60, bgPaint);
          
          // 텍스트
          const textPaint = Paint();
          textPaint.color = selected.color;
          textPaint.fontSize = 24;
          textPaint.fontWeight = 'bold';
          canvas.drawText(
            `${percentage}%`,
            { x: centerX - 30, y: centerY - 10 },
            textPaint
          );
          
          textPaint.fontSize = 14;
          textPaint.color = '#495057';
          canvas.drawText(
            selected.label,
            { x: centerX - 20, y: centerY + 15 },
            textPaint
          );
        }
      }
    };
    
    // 마우스 위치로 파이 조각 찾기
    const findPieSlice = (x, y, centerX, centerY) => {
      const dx = x - centerX;
      const dy = y - centerY;
      const distance = Math.sqrt(dx * dx + dy * dy);
      
      if (distance > 150) return -1;
      
      let angle = Math.atan2(dy, dx) + Math.PI / 2;
      if (angle < 0) angle += Math.PI * 2;
      
      let currentAngle = 0;
      for (let i = 0; i < data.length; i++) {
        const sweepAngle = (data[i].value / total) * Math.PI * 2;
        if (angle >= currentAngle && angle < currentAngle + sweepAngle) {
          return i;
        }
        currentAngle += sweepAngle;
      }
      return -1;
    };
    
    return Container({
      width: 450,
      height: 400,
      padding: EdgeInsets.all(20),
      color: 'white',
      child: Column({
        children: [
          Row({
            mainAxisAlignment: 'spaceBetween',
            children: [
              Text('월별 지출 내역', {
                style: TextStyle({
                  fontSize: 20,
                  fontWeight: 'bold',
                  color: '#212529'
                })
              }),
              Text(`총 ${(total/1000).toFixed(0)}만원`, {
                style: TextStyle({
                  fontSize: 16,
                  color: '#868e96'
                })
              })
            ]
          }),
          SizedBox({ height: 20 }),
          Container({
            height: 250,
            child: GestureDetector({
              onTapDown: (e) => {
                const rect = e.currentTarget.getBoundingClientRect();
                const x = e.clientX - rect.left;
                const y = e.clientY - rect.top;
                const index = findPieSlice(x, y, rect.width/2, rect.height/2);
                if (index >= 0) {
                  setSelectedIndex(index);
                }
              },
              onPointerMove: (e) => {
                const rect = e.currentTarget.getBoundingClientRect();
                const x = e.clientX - rect.left;
                const y = e.clientY - rect.top;
                const index = findPieSlice(x, y, rect.width/2, rect.height/2);
                setHoveredIndex(index);
              },
              onPointerLeave: () => setHoveredIndex(null),
              child: CustomPaint({
                painter: InteractivePieChartPainter
              })
            })
          }),
          SizedBox({ height: 20 }),
          // 범례
          Container({
            child: Wrap({
              spacing: 16,
              runSpacing: 8,
              children: data.map((item, index) => 
                GestureDetector({
                  onTap: () => setSelectedIndex(index),
                  child: Container({
                    padding: EdgeInsets.symmetric({ horizontal: 12, vertical: 6 }),
                    decoration: {
                      color: selectedIndex === index ? item.color : 'white',
                      borderRadius: BorderRadius.circular(16),
                      border: {
                        color: item.color,
                        width: 1
                      }
                    },
                    child: Row({
                      mainAxisSize: 'min',
                      children: [
                        Container({
                          width: 8,
                          height: 8,
                          decoration: {
                            color: selectedIndex === index ? 'white' : item.color,
                            shape: 'circle'
                          }
                        }),
                        SizedBox({ width: 6 }),
                        Text(item.label, {
                          style: TextStyle({
                            fontSize: 12,
                            color: selectedIndex === index ? 'white' : '#495057'
                          })
                        }),
                        SizedBox({ width: 4 }),
                        Text(`${(item.value/10000).toFixed(0)}만`, {
                          style: TextStyle({
                            fontSize: 12,
                            fontWeight: 'bold',
                            color: selectedIndex === index ? 'white' : item.color
                          })
                        })
                      ]
                    })
                  })
                })
              )
            })
          })
        ]
      })
    });
  })()}
</Widget>

## 애니메이션 파이 차트

부드러운 진입 애니메이션이 있는 파이 차트입니다.

<Widget name="cookbook-animated-pie-chart">
  {(() => {
    const [animationProgress, setAnimationProgress] = React.useState(0);
    const [showLabels, setShowLabels] = React.useState(false);
    
    React.useEffect(() => {
      const startTime = Date.now();
      const duration = 1000;
      
      const animate = () => {
        const elapsed = Date.now() - startTime;
        const progress = Math.min(elapsed / duration, 1);
        
        // Ease-out 곡선
        const eased = 1 - Math.pow(1 - progress, 3);
        setAnimationProgress(eased);
        
        if (progress < 1) {
          requestAnimationFrame(animate);
        } else {
          setShowLabels(true);
        }
      };
      
      requestAnimationFrame(animate);
    }, []);
    
    const data = [
      { category: '개발', hours: 8, color: '#4c6ef5' },
      { category: '회의', hours: 2, color: '#51cf66' },
      { category: '문서화', hours: 3, color: '#ff922b' },
      { category: '학습', hours: 2, color: '#845ef7' },
      { category: '휴식', hours: 1, color: '#20c997' }
    ];
    
    const total = data.reduce((sum, item) => sum + item.hours, 0);
    
    const AnimatedPieChartPainter = {
      paint: (canvas, size) => {
        const centerX = size.width / 2;
        const centerY = size.height / 2;
        const radius = Math.min(centerX, centerY) - 30;
        
        // 배경 원
        const bgPaint = Paint();
        bgPaint.color = '#f1f3f5';
        bgPaint.style = 'fill';
        canvas.drawCircle({ x: centerX, y: centerY }, radius, bgPaint);
        
        let startAngle = -Math.PI / 2;
        const totalSweep = Math.PI * 2 * animationProgress;
        let currentSweep = 0;
        
        data.forEach((item, index) => {
          const itemSweep = (item.hours / total) * Math.PI * 2;
          const actualSweep = Math.min(itemSweep, totalSweep - currentSweep);
          
          if (actualSweep > 0) {
            const paint = Paint();
            paint.color = item.color;
            paint.style = 'fill';
            
            const path = Path();
            path.moveTo(centerX, centerY);
            path.arcTo(
              Rect.fromCircle({ x: centerX, y: centerY }, radius),
              startAngle,
              actualSweep,
              false
            );
            path.close();
            
            canvas.drawPath(path, paint);
            
            // 애니메이션 완료 후 레이블 표시
            if (showLabels && actualSweep === itemSweep) {
              const midAngle = startAngle + itemSweep / 2;
              const labelRadius = radius + 20;
              const labelX = centerX + labelRadius * Math.cos(midAngle);
              const labelY = centerY + labelRadius * Math.sin(midAngle);
              
              // 연결선
              const linePaint = Paint();
              linePaint.color = item.color;
              linePaint.strokeWidth = 2;
              linePaint.style = 'stroke';
              
              const lineStartRadius = radius + 5;
              const lineStartX = centerX + lineStartRadius * Math.cos(midAngle);
              const lineStartY = centerY + lineStartRadius * Math.sin(midAngle);
              
              canvas.drawLine(
                { x: lineStartX, y: lineStartY },
                { x: labelX, y: labelY },
                linePaint
              );
              
              // 레이블 배경
              const bgPaint = Paint();
              bgPaint.color = item.color;
              bgPaint.style = 'fill';
              
              const labelText = `${item.hours}h`;
              const labelWidth = 30;
              const labelHeight = 20;
              
              canvas.drawRRect(
                RRect.fromRectAndRadius(
                  Rect.fromLTWH(
                    labelX - labelWidth/2,
                    labelY - labelHeight/2,
                    labelWidth,
                    labelHeight
                  ),
                  Radius.circular(10)
                ),
                bgPaint
              );
              
              // 레이블 텍스트
              const textPaint = Paint();
              textPaint.color = 'white';
              textPaint.fontSize = 12;
              textPaint.fontWeight = 'bold';
              
              canvas.drawText(
                labelText,
                { x: labelX - 10, y: labelY - 5 },
                textPaint
              );
            }
          }
          
          startAngle += itemSweep;
          currentSweep += actualSweep;
        });
        
        // 중앙 원 (도넛 효과)
        const centerPaint = Paint();
        centerPaint.color = 'white';
        centerPaint.style = 'fill';
        canvas.drawCircle({ x: centerX, y: centerY }, radius * 0.4, centerPaint);
        
        // 중앙 텍스트
        if (animationProgress === 1) {
          const textPaint = Paint();
          textPaint.color = '#212529';
          textPaint.fontSize = 24;
          textPaint.fontWeight = 'bold';
          canvas.drawText(
            `${total}h`,
            { x: centerX - 20, y: centerY - 10 },
            textPaint
          );
          
          textPaint.fontSize = 12;
          textPaint.color = '#868e96';
          textPaint.fontWeight = 'normal';
          canvas.drawText(
            '총 시간',
            { x: centerX - 20, y: centerY + 10 },
            textPaint
          );
        }
      }
    };
    
    return Container({
      width: 400,
      height: 400,
      padding: EdgeInsets.all(20),
      color: '#f8f9fa',
      child: Column({
        children: [
          Text('일일 시간 분배', {
            style: TextStyle({
              fontSize: 20,
              fontWeight: 'bold',
              color: '#212529'
            })
          }),
          SizedBox({ height: 20 }),
          Expanded({
            child: CustomPaint({
              painter: AnimatedPieChartPainter
            })
          }),
          // 카테고리 리스트
          AnimatedOpacity({
            opacity: showLabels ? 1 : 0,
            duration: 500,
            child: Container({
              padding: EdgeInsets.only({ top: 20 }),
              child: Row({
                mainAxisAlignment: 'spaceEvenly',
                children: data.map(item => 
                  Column({
                    children: [
                      Container({
                        width: 16,
                        height: 16,
                        decoration: {
                          color: item.color,
                          shape: 'circle'
                        }
                      }),
                      SizedBox({ height: 4 }),
                      Text(item.category, {
                        style: TextStyle({
                          fontSize: 12,
                          color: '#495057'
                        })
                      })
                    ]
                  })
                )
              })
            })
          })
        ]
      })
    });
  })()}
</Widget>

## 구현 핵심

### 1. 각도 계산
```typescript
const startAngle = -Math.PI / 2; // 12시 방향
const sweepAngle = (value / total) * Math.PI * 2;
```

### 2. 파이 조각 그리기
```typescript
const path = Path();
path.moveTo(centerX, centerY);
path.arcTo(
  Rect.fromCircle({ x: centerX, y: centerY }, radius),
  startAngle,
  sweepAngle,
  false
);
path.close();
```

### 3. 마우스 위치로 조각 찾기
```typescript
const angle = Math.atan2(y - centerY, x - centerX);
// 각도를 0-2π 범위로 정규화하고 해당 조각 찾기
```

### 4. 레이블 위치 계산
```typescript
const midAngle = startAngle + sweepAngle / 2;
const labelX = centerX + radius * Math.cos(midAngle);
const labelY = centerY + radius * Math.sin(midAngle);
```

## 고급 기능

1. **드릴다운**: 조각 클릭 시 상세 데이터 표시
2. **동적 업데이트**: 실시간 데이터 반영
3. **3D 효과**: 그림자와 하이라이트로 입체감
4. **툴팁**: 마우스 오버 시 상세 정보
5. **범례 토글**: 특정 카테고리 숨기기/보이기

## 성능 최적화

1. **경로 캐싱**: 자주 그리는 경로는 미리 계산
2. **히트 테스트 최적화**: 영역 사전 계산
3. **애니메이션 프레임**: requestAnimationFrame 사용
