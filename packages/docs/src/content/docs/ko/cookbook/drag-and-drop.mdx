import { Widget } from '@meursyphus/flitter-svelte';
import { Container, Stack, Positioned, Draggable, Text, TextStyle, Center, EdgeInsets, BorderRadius, BoxShadow, Column, Row } from '@meursyphus/flitter';

# 드래그 앤 드롭

드래그 앤 드롭은 직관적인 사용자 인터랙션을 제공하는 중요한 기능입니다. Flitter의 `Draggable` 위젯을 사용하여 쉽게 구현할 수 있습니다.

## 기본 드래그 가능한 위젯

가장 간단한 형태의 드래그 가능한 위젯입니다.

<Widget name="cookbook-drag-basic">
  {(() => {
    const [position, setPosition] = React.useState({ x: 100, y: 100 });
    
    return Container({
      width: 400,
      height: 300,
      color: '#f8f9fa',
      child: Stack({
        children: [
          Positioned({
            left: position.x,
            top: position.y,
            child: Draggable({
              onDragEnd: (details) => {
                setPosition({
                  x: position.x + details.offset.dx,
                  y: position.y + details.offset.dy
                });
              },
              feedback: Container({
                width: 100,
                height: 100,
                decoration: {
                  color: '#339af0',
                  borderRadius: BorderRadius.circular(12),
                  boxShadow: [
                    BoxShadow({
                      color: 'rgba(0,0,0,0.3)',
                      blurRadius: 10,
                      offset: { x: 0, y: 5 }
                    })
                  ]
                },
                child: Center({
                  child: Text('드래그 중', {
                    style: TextStyle({
                      color: 'white',
                      fontWeight: 'bold'
                    })
                  })
                })
              }),
              child: Container({
                width: 100,
                height: 100,
                decoration: {
                  color: '#4c6ef5',
                  borderRadius: BorderRadius.circular(12)
                },
                child: Center({
                  child: Text('드래그하세요', {
                    style: TextStyle({
                      color: 'white',
                      fontWeight: 'bold'
                    })
                  })
                })
              })
            })
          })
        ]
      })
    });
  })()}
</Widget>

```typescript
Draggable({
  onDragEnd: (details) => {
    // 드래그 종료 시 위치 업데이트
    setPosition({
      x: position.x + details.offset.dx,
      y: position.y + details.offset.dy
    });
  },
  feedback: // 드래그 중 표시될 위젯,
  child: // 원본 위젯
})
```

## 드래그 타겟과 함께 사용

특정 영역에 드롭할 수 있는 드래그 앤 드롭입니다.

<Widget name="cookbook-drag-target">
  {(() => {
    const [items, setItems] = React.useState(['사과', '바나나', '오렌지']);
    const [basket, setBasket] = React.useState([]);
    const [draggedItem, setDraggedItem] = React.useState(null);
    
    return Container({
      width: 400,
      height: 400,
      padding: EdgeInsets.all(20),
      color: '#f8f9fa',
      child: Column({
        children: [
          Text('과일을 바구니로 드래그하세요', {
            style: TextStyle({
              fontSize: 18,
              fontWeight: 'bold',
              color: '#495057'
            })
          }),
          SizedBox({ height: 20 }),
          Row({
            mainAxisAlignment: 'spaceEvenly',
            children: items.map(item => 
              Draggable({
                data: item,
                onDragStarted: () => setDraggedItem(item),
                onDragEnd: (_) => setDraggedItem(null),
                feedback: Container({
                  padding: EdgeInsets.all(12),
                  decoration: {
                    color: '#ff6b6b',
                    borderRadius: BorderRadius.circular(8),
                    boxShadow: [
                      BoxShadow({
                        color: 'rgba(0,0,0,0.3)',
                        blurRadius: 8,
                        offset: { x: 0, y: 4 }
                      })
                    ]
                  },
                  child: Text(item, {
                    style: TextStyle({
                      color: 'white',
                      fontWeight: 'bold'
                    })
                  })
                }),
                child: Container({
                  padding: EdgeInsets.all(12),
                  margin: EdgeInsets.all(4),
                  decoration: {
                    color: '#ffa94d',
                    borderRadius: BorderRadius.circular(8)
                  },
                  child: Text(item, {
                    style: TextStyle({
                      color: 'white',
                      fontWeight: 'bold'
                    })
                  })
                })
              })
            )
          }),
          SizedBox({ height: 40 }),
          DragTarget({
            onAccept: (data) => {
              setBasket([...basket, data]);
              setItems(items.filter(item => item !== data));
            },
            builder: (context, candidateData, rejectedData) => {
              const isHovering = candidateData.length > 0;
              
              return Container({
                width: 300,
                height: 150,
                decoration: {
                  color: isHovering ? '#e7f5ff' : 'white',
                  borderRadius: BorderRadius.circular(16),
                  border: {
                    color: isHovering ? '#339af0' : '#dee2e6',
                    width: 2,
                    style: 'dashed'
                  }
                },
                child: Center({
                  child: basket.length === 0
                    ? Text('바구니 (여기에 드롭하세요)', {
                        style: TextStyle({
                          color: '#868e96',
                          fontSize: 16
                        })
                      })
                    : Column({
                        mainAxisAlignment: 'center',
                        children: [
                          Text('바구니 내용:', {
                            style: TextStyle({
                              fontWeight: 'bold',
                              color: '#495057'
                            })
                          }),
                          SizedBox({ height: 10 }),
                          ...basket.map(item => 
                            Text(`• ${item}`, {
                              style: TextStyle({
                                color: '#495057'
                              })
                            })
                          )
                        ]
                      })
                })
              });
            }
          })
        ]
      })
    });
  })()}
</Widget>

## 정렬 가능한 리스트

드래그로 항목 순서를 변경할 수 있는 리스트입니다.

<Widget name="cookbook-drag-reorder">
  {(() => {
    const [tasks, setTasks] = React.useState([
      { id: 1, text: '아침 운동하기', color: '#ff6b6b' },
      { id: 2, text: '이메일 확인하기', color: '#4c6ef5' },
      { id: 3, text: '프로젝트 작업하기', color: '#51cf66' },
      { id: 4, text: '회의 참석하기', color: '#ff922b' },
      { id: 5, text: '보고서 작성하기', color: '#845ef7' }
    ]);
    
    const moveItem = (fromIndex, toIndex) => {
      const newTasks = [...tasks];
      const [removed] = newTasks.splice(fromIndex, 1);
      newTasks.splice(toIndex, 0, removed);
      setTasks(newTasks);
    };
    
    return Container({
      width: 350,
      padding: EdgeInsets.all(20),
      color: '#f8f9fa',
      child: Column({
        children: [
          Text('할 일 목록 (드래그로 순서 변경)', {
            style: TextStyle({
              fontSize: 18,
              fontWeight: 'bold',
              color: '#495057'
            })
          }),
          SizedBox({ height: 20 }),
          ...tasks.map((task, index) => 
            Container({
              key: task.id,
              margin: EdgeInsets.only({ bottom: 8 }),
              child: Draggable({
                data: index,
                feedback: Container({
                  width: 300,
                  padding: EdgeInsets.all(16),
                  decoration: {
                    color: task.color,
                    borderRadius: BorderRadius.circular(8),
                    boxShadow: [
                      BoxShadow({
                        color: 'rgba(0,0,0,0.3)',
                        blurRadius: 10,
                        offset: { x: 0, y: 5 }
                      })
                    ]
                  },
                  child: Text(task.text, {
                    style: TextStyle({
                      color: 'white',
                      fontSize: 16,
                      fontWeight: 'bold'
                    })
                  })
                }),
                child: DragTarget({
                  onAccept: (fromIndex) => {
                    if (fromIndex !== index) {
                      moveItem(fromIndex, index);
                    }
                  },
                  builder: (context, candidateData, rejectedData) => {
                    return Container({
                      width: double.infinity,
                      padding: EdgeInsets.all(16),
                      decoration: {
                        color: task.color,
                        borderRadius: BorderRadius.circular(8)
                      },
                      child: Row({
                        children: [
                          Text('☰ ', {
                            style: TextStyle({
                              color: 'white',
                              fontSize: 18
                            })
                          }),
                          Text(task.text, {
                            style: TextStyle({
                              color: 'white',
                              fontSize: 16
                            })
                          })
                        ]
                      })
                    });
                  }
                })
              })
            })
          )
        ]
      })
    });
  })()}
</Widget>

## 제한된 영역 내 드래그

특정 영역 내에서만 드래그가 가능하도록 제한하는 예제입니다.

<Widget name="cookbook-drag-constrained">
  {(() => {
    const [position, setPosition] = React.useState({ x: 150, y: 150 });
    const containerSize = { width: 400, height: 300 };
    const boxSize = { width: 80, height: 80 };
    
    const constrainPosition = (x, y) => {
      const maxX = containerSize.width - boxSize.width;
      const maxY = containerSize.height - boxSize.height;
      
      return {
        x: Math.max(0, Math.min(x, maxX)),
        y: Math.max(0, Math.min(y, maxY))
      };
    };
    
    return Container({
      width: containerSize.width,
      height: containerSize.height,
      decoration: {
        color: '#e9ecef',
        border: {
          color: '#495057',
          width: 2
        }
      },
      child: Stack({
        children: [
          Positioned({
            left: position.x,
            top: position.y,
            child: Draggable({
              onDragEnd: (details) => {
                const newPosition = constrainPosition(
                  position.x + details.offset.dx,
                  position.y + details.offset.dy
                );
                setPosition(newPosition);
              },
              feedback: Container({
                width: boxSize.width,
                height: boxSize.height,
                decoration: {
                  color: '#228be6',
                  borderRadius: BorderRadius.circular(40),
                  boxShadow: [
                    BoxShadow({
                      color: 'rgba(0,0,0,0.3)',
                      blurRadius: 10,
                      offset: { x: 0, y: 5 }
                    })
                  ]
                },
                child: Center({
                  child: Text('🎯', {
                    style: TextStyle({ fontSize: 32 })
                  })
                })
              }),
              child: Container({
                width: boxSize.width,
                height: boxSize.height,
                decoration: {
                  color: '#339af0',
                  borderRadius: BorderRadius.circular(40)
                },
                child: Center({
                  child: Text('🎯', {
                    style: TextStyle({ fontSize: 32 })
                  })
                })
              })
            })
          }),
          Positioned({
            bottom: 10,
            left: 10,
            child: Text('영역 내에서만 이동 가능', {
              style: TextStyle({
                color: '#868e96',
                fontSize: 14
              })
            })
          })
        ]
      })
    });
  })()}
</Widget>

## 핵심 개념

### 1. Draggable 위젯
```typescript
Draggable({
  data: myData,              // 드래그할 데이터
  child: Widget(),           // 원본 위젯
  feedback: Widget(),        // 드래그 중 표시될 위젯
  onDragStarted: () => {},   // 드래그 시작 시
  onDragEnd: (details) => {} // 드래그 종료 시
})
```

### 2. DragTarget 위젯
```typescript
DragTarget({
  onAccept: (data) => {},    // 드롭 허용 시
  onWillAccept: (data) => {  // 드롭 가능 여부 판단
    return true;
  },
  builder: (context, candidateData, rejectedData) => {
    // candidateData: 현재 hover 중인 데이터
    return Widget();
  }
})
```

### 3. 드래그 피드백 커스터마이징
```typescript
feedback: Transform({
  transform: Matrix4.rotationZ(0.1), // 약간 회전
  child: Opacity({
    opacity: 0.8,
    child: YourWidget()
  })
})
```

## 성능 최적화

1. **복잡한 위젯**: feedback 위젯은 단순하게 유지하세요
2. **상태 업데이트**: 드래그 중에는 최소한의 상태만 업데이트
3. **제약 조건**: 미리 계산하여 드래그 중 계산 최소화

## 접근성 고려사항

1. 키보드로도 조작 가능하도록 대체 방법 제공
2. 스크린 리더를 위한 적절한 라벨 추가
3. 드래그 가능한 요소임을 시각적으로 표시
