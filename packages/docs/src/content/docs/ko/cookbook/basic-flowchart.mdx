import { Widget } from '@meursyphus/flitter-svelte';
import { Container, Stack, Positioned, CustomPaint, GestureDetector, Text, TextStyle, Center, EdgeInsets, BorderRadius, BoxShadow } from '@meursyphus/flitter';

# 기본 플로우차트

플로우차트는 프로세스나 시스템의 흐름을 시각적으로 표현하는 다이어그램입니다. Flitter로 인터랙티브한 플로우차트를 구현해봅시다.

## 간단한 플로우차트

기본적인 노드와 연결선으로 구성된 플로우차트입니다.

<Widget name="cookbook-basic-flowchart">
  {(() => {
    const nodes = [
      { id: 'start', x: 200, y: 50, width: 120, height: 50, label: '시작', type: 'ellipse', color: '#51cf66' },
      { id: 'process1', x: 200, y: 150, width: 120, height: 60, label: '데이터 입력', type: 'rect', color: '#4c6ef5' },
      { id: 'decision', x: 200, y: 260, width: 120, height: 80, label: '유효한가?', type: 'diamond', color: '#ff922b' },
      { id: 'process2', x: 80, y: 380, width: 100, height: 60, label: '에러 처리', type: 'rect', color: '#ff6b6b' },
      { id: 'process3', x: 320, y: 380, width: 100, height: 60, label: '처리 진행', type: 'rect', color: '#4c6ef5' },
      { id: 'end', x: 200, y: 500, width: 120, height: 50, label: '종료', type: 'ellipse', color: '#868e96' }
    ];
    
    const connections = [
      { from: 'start', to: 'process1' },
      { from: 'process1', to: 'decision' },
      { from: 'decision', to: 'process2', label: '아니오' },
      { from: 'decision', to: 'process3', label: '예' },
      { from: 'process2', to: 'end' },
      { from: 'process3', to: 'end' }
    ];
    
    const getNodeCenter = (node) => ({
      x: node.x + node.width / 2,
      y: node.y + node.height / 2
    });
    
    const ConnectionPainter = {
      paint: (canvas, size) => {
        const paint = Paint();
        paint.color = '#495057';
        paint.strokeWidth = 2;
        paint.style = 'stroke';
        
        connections.forEach(conn => {
          const fromNode = nodes.find(n => n.id === conn.from);
          const toNode = nodes.find(n => n.id === conn.to);
          
          if (fromNode && toNode) {
            const from = getNodeCenter(fromNode);
            const to = getNodeCenter(toNode);
            
            // 연결선 그리기
            canvas.drawLine(
              { x: from.x, y: fromNode.y + fromNode.height },
              { x: to.x, y: toNode.y },
              paint
            );
            
            // 화살표 그리기
            const arrowSize = 8;
            const angle = Math.atan2(to.y - from.y, to.x - from.x);
            
            paint.style = 'fill';
            canvas.drawPath(
              Path()
                ..moveTo(to.x, toNode.y)
                ..lineTo(
                  to.x - arrowSize * Math.cos(angle - Math.PI / 6),
                  toNode.y - arrowSize * Math.sin(angle - Math.PI / 6)
                )
                ..lineTo(
                  to.x - arrowSize * Math.cos(angle + Math.PI / 6),
                  toNode.y - arrowSize * Math.sin(angle + Math.PI / 6)
                )
                ..close(),
              paint
            );
            
            // 레이블 그리기
            if (conn.label) {
              const textPaint = Paint();
              textPaint.color = '#868e96';
              textPaint.fontSize = 12;
              
              const midX = (from.x + to.x) / 2;
              const midY = (from.y + to.y) / 2;
              
              canvas.drawText(
                conn.label,
                { x: midX - 20, y: midY },
                textPaint
              );
            }
          }
        });
      }
    };
    
    return Container({
      width: 500,
      height: 600,
      color: '#f8f9fa',
      child: Stack({
        children: [
          // 연결선 레이어
          CustomPaint({
            painter: ConnectionPainter
          }),
          // 노드 레이어
          ...nodes.map(node => 
            Positioned({
              left: node.x,
              top: node.y,
              child: Container({
                width: node.width,
                height: node.height,
                decoration: {
                  color: node.color,
                  borderRadius: node.type === 'ellipse' ? BorderRadius.circular(25) : 
                                node.type === 'diamond' ? BorderRadius.circular(8) : 
                                BorderRadius.circular(4),
                  boxShadow: [
                    BoxShadow({
                      color: 'rgba(0,0,0,0.1)',
                      blurRadius: 4,
                      offset: { x: 0, y: 2 }
                    })
                  ]
                },
                child: Center({
                  child: Text(node.label, {
                    style: TextStyle({
                      color: 'white',
                      fontSize: 14,
                      fontWeight: 'bold'
                    })
                  })
                })
              })
            })
          )
        ]
      })
    });
  })()}
</Widget>

## 인터랙티브 플로우차트

클릭 가능한 노드와 하이라이트 효과가 있는 플로우차트입니다.

<Widget name="cookbook-interactive-flowchart">
  {(() => {
    const [selectedNode, setSelectedNode] = React.useState(null);
    const [hoveredNode, setHoveredNode] = React.useState(null);
    
    const workflow = {
      nodes: [
        { id: 'input', x: 50, y: 100, label: '데이터 수집', icon: '📊' },
        { id: 'clean', x: 200, y: 100, label: '데이터 정제', icon: '🧹' },
        { id: 'analyze', x: 350, y: 100, label: '분석', icon: '🔍' },
        { id: 'ml', x: 200, y: 250, label: '머신러닝', icon: '🤖' },
        { id: 'visualize', x: 350, y: 250, label: '시각화', icon: '📈' },
        { id: 'report', x: 275, y: 400, label: '보고서', icon: '📄' }
      ],
      edges: [
        { from: 'input', to: 'clean' },
        { from: 'clean', to: 'analyze' },
        { from: 'clean', to: 'ml' },
        { from: 'analyze', to: 'visualize' },
        { from: 'ml', to: 'report' },
        { from: 'visualize', to: 'report' }
      ]
    };
    
    const isConnected = (nodeId) => {
      if (!selectedNode) return false;
      return workflow.edges.some(edge => 
        (edge.from === selectedNode && edge.to === nodeId) ||
        (edge.to === selectedNode && edge.from === nodeId)
      );
    };
    
    const EdgePainter = {
      paint: (canvas, size) => {
        const paint = Paint();
        
        workflow.edges.forEach(edge => {
          const fromNode = workflow.nodes.find(n => n.id === edge.from);
          const toNode = workflow.nodes.find(n => n.id === edge.to);
          
          if (fromNode && toNode) {
            const isHighlighted = selectedNode && 
              (edge.from === selectedNode || edge.to === selectedNode);
            
            paint.color = isHighlighted ? '#4c6ef5' : '#dee2e6';
            paint.strokeWidth = isHighlighted ? 3 : 2;
            paint.style = 'stroke';
            
            const fromX = fromNode.x + 60;
            const fromY = fromNode.y + 40;
            const toX = toNode.x + 60;
            const toY = toNode.y + 40;
            
            // 베지어 곡선으로 연결
            const path = Path();
            path.moveTo(fromX, fromY);
            
            const controlY = (fromY + toY) / 2 + 30;
            path.quadraticBezierTo(
              fromX, controlY,
              toX, toY
            );
            
            canvas.drawPath(path, paint);
            
            // 화살표
            const t = 0.95; // 곡선 상의 위치
            const x = (1-t)*(1-t)*fromX + 2*(1-t)*t*fromX + t*t*toX;
            const y = (1-t)*(1-t)*fromY + 2*(1-t)*t*controlY + t*t*toY;
            
            const angle = Math.atan2(toY - y, toX - x);
            const arrowSize = 8;
            
            paint.style = 'fill';
            const arrowPath = Path();
            arrowPath.moveTo(toX, toY);
            arrowPath.lineTo(
              toX - arrowSize * Math.cos(angle - Math.PI/6),
              toY - arrowSize * Math.sin(angle - Math.PI/6)
            );
            arrowPath.lineTo(
              toX - arrowSize * Math.cos(angle + Math.PI/6),
              toY - arrowSize * Math.sin(angle + Math.PI/6)
            );
            arrowPath.close();
            
            canvas.drawPath(arrowPath, paint);
          }
        });
      }
    };
    
    return Container({
      width: 500,
      height: 500,
      color: 'white',
      padding: EdgeInsets.all(20),
      child: Stack({
        children: [
          // 엣지 레이어
          CustomPaint({
            painter: EdgePainter
          }),
          // 노드 레이어
          ...workflow.nodes.map(node => {
            const isSelected = selectedNode === node.id;
            const isHovered = hoveredNode === node.id;
            const isRelated = isConnected(node.id);
            
            return Positioned({
              left: node.x,
              top: node.y,
              child: GestureDetector({
                onTap: () => setSelectedNode(
                  selectedNode === node.id ? null : node.id
                ),
                onEnter: () => setHoveredNode(node.id),
                onExit: () => setHoveredNode(null),
                child: AnimatedContainer({
                  duration: 200,
                  width: 120,
                  height: 80,
                  decoration: {
                    color: isSelected ? '#4c6ef5' : 
                           isRelated ? '#e7f5ff' : 
                           'white',
                    borderRadius: BorderRadius.circular(12),
                    border: {
                      color: isSelected ? '#4c6ef5' : 
                             isHovered ? '#868e96' : 
                             '#e9ecef',
                      width: 2
                    },
                    boxShadow: [
                      BoxShadow({
                        color: isSelected ? 'rgba(76,110,245,0.2)' : 
                               isHovered ? 'rgba(0,0,0,0.1)' : 
                               'rgba(0,0,0,0.05)',
                        blurRadius: isSelected ? 12 : 6,
                        offset: { x: 0, y: isSelected ? 6 : 3 }
                      })
                    ]
                  },
                  child: Column({
                    mainAxisAlignment: 'center',
                    children: [
                      Text(node.icon, {
                        style: TextStyle({ fontSize: 24 })
                      }),
                      SizedBox({ height: 4 }),
                      Text(node.label, {
                        style: TextStyle({
                          fontSize: 12,
                          fontWeight: isSelected ? 'bold' : 'normal',
                          color: isSelected ? 'white' : '#495057'
                        })
                      })
                    ]
                  })
                })
              })
            });
          }),
          // 정보 패널
          if (selectedNode) Positioned({
            bottom: 0,
            left: 0,
            right: 0,
            child: Container({
              padding: EdgeInsets.all(16),
              decoration: {
                color: '#f8f9fa',
                borderRadius: BorderRadius.circular(8),
                border: {
                  color: '#e9ecef',
                  width: 1
                }
              },
              child: Text(
                `선택된 노드: ${workflow.nodes.find(n => n.id === selectedNode)?.label}`,
                {
                  style: TextStyle({
                    fontSize: 14,
                    color: '#495057'
                  })
                }
              )
            })
          })
        ]
      })
    });
  })()}
</Widget>

## 플로우차트 빌더

드래그로 노드를 추가하고 연결할 수 있는 빌더입니다.

<Widget name="cookbook-flowchart-builder">
  {(() => {
    const [nodes, setNodes] = React.useState([
      { id: '1', x: 100, y: 100, type: 'start', label: '시작' }
    ]);
    const [edges, setEdges] = React.useState([]);
    const [selectedTool, setSelectedTool] = React.useState('select');
    const [connecting, setConnecting] = React.useState(null);
    
    const nodeTypes = [
      { type: 'process', label: '프로세스', color: '#4c6ef5', icon: '⚙️' },
      { type: 'decision', label: '결정', color: '#ff922b', icon: '❓' },
      { type: 'data', label: '데이터', color: '#51cf66', icon: '📊' },
      { type: 'end', label: '종료', color: '#868e96', icon: '🏁' }
    ];
    
    const handleCanvasClick = (e) => {
      if (selectedTool !== 'select' && selectedTool !== 'connect') {
        const rect = e.currentTarget.getBoundingClientRect();
        const x = e.clientX - rect.left - 50;
        const y = e.clientY - rect.top - 30;
        
        const nodeType = nodeTypes.find(t => t.type === selectedTool);
        if (nodeType) {
          const newNode = {
            id: Date.now().toString(),
            x, y,
            type: selectedTool,
            label: nodeType.label,
            color: nodeType.color,
            icon: nodeType.icon
          };
          setNodes([...nodes, newNode]);
        }
      }
    };
    
    const handleNodeClick = (nodeId) => {
      if (selectedTool === 'connect') {
        if (!connecting) {
          setConnecting(nodeId);
        } else if (connecting !== nodeId) {
          setEdges([...edges, { from: connecting, to: nodeId }]);
          setConnecting(null);
        }
      }
    };
    
    return Container({
      width: 550,
      height: 500,
      color: 'white',
      child: Column({
        children: [
          // 툴바
          Container({
            padding: EdgeInsets.all(12),
            color: '#f8f9fa',
            child: Row({
              children: [
                GestureDetector({
                  onTap: () => setSelectedTool('select'),
                  child: Container({
                    padding: EdgeInsets.all(8),
                    margin: EdgeInsets.only({ right: 8 }),
                    decoration: {
                      color: selectedTool === 'select' ? '#4c6ef5' : 'white',
                      borderRadius: BorderRadius.circular(4)
                    },
                    child: Text('👆', { style: TextStyle({ fontSize: 20 }) })
                  })
                }),
                GestureDetector({
                  onTap: () => setSelectedTool('connect'),
                  child: Container({
                    padding: EdgeInsets.all(8),
                    margin: EdgeInsets.only({ right: 16 }),
                    decoration: {
                      color: selectedTool === 'connect' ? '#4c6ef5' : 'white',
                      borderRadius: BorderRadius.circular(4)
                    },
                    child: Text('🔗', { style: TextStyle({ fontSize: 20 }) })
                  })
                }),
                ...nodeTypes.map(nodeType => 
                  GestureDetector({
                    onTap: () => setSelectedTool(nodeType.type),
                    child: Container({
                      padding: EdgeInsets.symmetric({ horizontal: 12, vertical: 8 }),
                      margin: EdgeInsets.only({ right: 8 }),
                      decoration: {
                        color: selectedTool === nodeType.type ? nodeType.color : 'white',
                        borderRadius: BorderRadius.circular(4),
                        border: {
                          color: nodeType.color,
                          width: 1
                        }
                      },
                      child: Row({
                        children: [
                          Text(nodeType.icon, { style: TextStyle({ fontSize: 16 }) }),
                          SizedBox({ width: 4 }),
                          Text(nodeType.label, {
                            style: TextStyle({
                              fontSize: 12,
                              color: selectedTool === nodeType.type ? 'white' : nodeType.color
                            })
                          })
                        ]
                      })
                    })
                  })
                )
              ]
            })
          }),
          // 캔버스
          Expanded({
            child: GestureDetector({
              onTapDown: handleCanvasClick,
              child: Container({
                color: '#fafafa',
                child: Stack({
                  children: [
                    // 엣지 레이어
                    CustomPaint({
                      painter: {
                        paint: (canvas, size) => {
                          const paint = Paint();
                          paint.color = '#495057';
                          paint.strokeWidth = 2;
                          paint.style = 'stroke';
                          
                          edges.forEach(edge => {
                            const fromNode = nodes.find(n => n.id === edge.from);
                            const toNode = nodes.find(n => n.id === edge.to);
                            
                            if (fromNode && toNode) {
                              canvas.drawLine(
                                { x: fromNode.x + 50, y: fromNode.y + 30 },
                                { x: toNode.x + 50, y: toNode.y + 30 },
                                paint
                              );
                            }
                          });
                        }
                      }
                    }),
                    // 노드 레이어
                    ...nodes.map(node => 
                      Positioned({
                        left: node.x,
                        top: node.y,
                        child: GestureDetector({
                          onTap: () => handleNodeClick(node.id),
                          child: Container({
                            width: 100,
                            height: 60,
                            decoration: {
                              color: node.color || '#4c6ef5',
                              borderRadius: BorderRadius.circular(8),
                              border: connecting === node.id ? {
                                color: '#212529',
                                width: 3
                              } : null,
                              boxShadow: [
                                BoxShadow({
                                  color: 'rgba(0,0,0,0.15)',
                                  blurRadius: 6,
                                  offset: { x: 0, y: 3 }
                                })
                              ]
                            },
                            child: Column({
                              mainAxisAlignment: 'center',
                              children: [
                                Text(node.icon || '📦', {
                                  style: TextStyle({ fontSize: 20 })
                                }),
                                Text(node.label, {
                                  style: TextStyle({
                                    color: 'white',
                                    fontSize: 12,
                                    fontWeight: 'bold'
                                  })
                                })
                              ]
                            })
                          })
                        })
                      })
                    ),
                    // 힌트 텍스트
                    if (nodes.length === 1) Positioned({
                      bottom: 20,
                      left: 20,
                      child: Container({
                        padding: EdgeInsets.all(12),
                        decoration: {
                          color: 'rgba(0,0,0,0.8)',
                          borderRadius: BorderRadius.circular(8)
                        },
                        child: Text(
                          selectedTool === 'select' ? '노드를 클릭하여 선택' :
                          selectedTool === 'connect' ? '노드를 클릭하여 연결' :
                          '캔버스를 클릭하여 노드 추가',
                          {
                            style: TextStyle({
                              color: 'white',
                              fontSize: 12
                            })
                          }
                        )
                      })
                    })
                  ]
                })
              })
            })
          })
        ]
      })
    });
  })()}
</Widget>

## 구현 핵심

### 1. 노드 구조
```typescript
interface Node {
  id: string;
  x: number;
  y: number;
  width: number;
  height: number;
  label: string;
  type: 'rect' | 'ellipse' | 'diamond';
  color: string;
}
```

### 2. 연결선 그리기
```typescript
// 직선 연결
canvas.drawLine(fromPoint, toPoint, paint);

// 베지어 곡선
const path = Path();
path.moveTo(from.x, from.y);
path.quadraticBezierTo(control.x, control.y, to.x, to.y);
canvas.drawPath(path, paint);
```

### 3. 충돌 감지
```typescript
const isPointInNode = (x, y, node) => {
  return x >= node.x && x <= node.x + node.width &&
         y >= node.y && y <= node.y + node.height;
};
```

## 고급 기능

1. **자동 레이아웃**: 노드 자동 배치 알고리즘
2. **그룹화**: 여러 노드를 그룹으로 묶기
3. **실행 취소/다시 실행**: 편집 히스토리
4. **내보내기**: SVG, PNG, JSON 형식으로 저장
5. **템플릿**: 미리 정의된 플로우차트 템플릿

## 성능 최적화

1. **레이어 분리**: 정적 요소와 동적 요소 분리
2. **뷰포트 컬링**: 보이는 영역만 렌더링
3. **디바운싱**: 드래그 이벤트 최적화
