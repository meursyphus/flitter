import { Widget } from '@meursyphus/flitter-svelte';
import { Container, Column, Row, Text, TextStyle, CustomPaint, GestureDetector, Stack, Positioned, EdgeInsets, BorderRadius, AnimatedContainer } from '@meursyphus/flitter';

# 간단한 바 차트

바 차트는 카테고리별 데이터를 비교할 때 가장 효과적인 시각화 방법입니다. CustomPaint를 사용하여 인터랙티브한 바 차트를 구현해봅시다.

## 기본 바 차트

가장 간단한 형태의 바 차트입니다.

<Widget name="cookbook-simple-bar-chart">
  {(() => {
    const data = [
      { label: '월', value: 30 },
      { label: '화', value: 45 },
      { label: '수', value: 38 },
      { label: '목', value: 52 },
      { label: '금', value: 41 },
      { label: '토', value: 35 },
      { label: '일', value: 28 }
    ];
    
    const maxValue = Math.max(...data.map(d => d.value));
    const chartHeight = 200;
    const barWidth = 40;
    const spacing = 10;
    
    return Container({
      width: 400,
      height: 300,
      padding: EdgeInsets.all(20),
      color: '#f8f9fa',
      child: Column({
        children: [
          Text('주간 판매량', {
            style: TextStyle({
              fontSize: 18,
              fontWeight: 'bold',
              color: '#212529'
            })
          }),
          SizedBox({ height: 20 }),
          Container({
            height: chartHeight,
            child: Row({
              mainAxisAlignment: 'spaceEvenly',
              crossAxisAlignment: 'end',
              children: data.map(item => {
                const height = (item.value / maxValue) * chartHeight;
                
                return Column({
                  mainAxisAlignment: 'end',
                  children: [
                    Container({
                      width: barWidth,
                      height: height,
                      decoration: {
                        color: '#4c6ef5',
                        borderRadius: BorderRadius.only({
                          topLeft: 4,
                          topRight: 4
                        })
                      }
                    }),
                    SizedBox({ height: 8 }),
                    Text(item.label, {
                      style: TextStyle({
                        fontSize: 12,
                        color: '#495057'
                      })
                    })
                  ]
                });
              })
            })
          })
        ]
      })
    });
  })()}
</Widget>

## 인터랙티브 바 차트

호버와 클릭 상호작용이 있는 바 차트입니다.

<Widget name="cookbook-interactive-bar-chart">
  {(() => {
    const [selectedIndex, setSelectedIndex] = React.useState(null);
    const [hoveredIndex, setHoveredIndex] = React.useState(null);
    
    const data = [
      { label: 'Q1', value: 65, color: '#ff6b6b' },
      { label: 'Q2', value: 78, color: '#4ecdc4' },
      { label: 'Q3', value: 92, color: '#45b7d1' },
      { label: 'Q4', value: 87, color: '#f9ca24' }
    ];
    
    const maxValue = 100;
    const chartHeight = 250;
    const barWidth = 60;
    
    return Container({
      width: 400,
      height: 350,
      padding: EdgeInsets.all(20),
      color: '#f8f9fa',
      child: Column({
        children: [
          Row({
            mainAxisAlignment: 'spaceBetween',
            children: [
              Text('분기별 성과', {
                style: TextStyle({
                  fontSize: 20,
                  fontWeight: 'bold',
                  color: '#212529'
                })
              }),
              if (selectedIndex !== null) Text(`${data[selectedIndex].value}%`, {
                style: TextStyle({
                  fontSize: 24,
                  fontWeight: 'bold',
                  color: data[selectedIndex].color
                })
              })
            ]
          }),
          SizedBox({ height: 30 }),
          Container({
            height: chartHeight,
            child: Stack({
              children: [
                // Y축 그리드 라인
                ...Array.from({ length: 5 }, (_, i) => {
                  const y = (i / 4) * chartHeight;
                  return Positioned({
                    top: y,
                    left: 0,
                    right: 0,
                    child: Container({
                      height: 1,
                      color: '#e9ecef'
                    })
                  });
                }),
                // 바 차트
                Row({
                  mainAxisAlignment: 'spaceEvenly',
                  crossAxisAlignment: 'end',
                  children: data.map((item, index) => {
                    const height = (item.value / maxValue) * chartHeight;
                    const isSelected = selectedIndex === index;
                    const isHovered = hoveredIndex === index;
                    
                    return GestureDetector({
                      onTap: () => setSelectedIndex(index),
                      onEnter: () => setHoveredIndex(index),
                      onExit: () => setHoveredIndex(null),
                      child: Column({
                        mainAxisAlignment: 'end',
                        children: [
                          // 값 표시
                          AnimatedOpacity({
                            opacity: isSelected || isHovered ? 1 : 0,
                            duration: 200,
                            child: Container({
                              margin: EdgeInsets.only({ bottom: 8 }),
                              padding: EdgeInsets.symmetric({ horizontal: 8, vertical: 4 }),
                              decoration: {
                                color: item.color,
                                borderRadius: BorderRadius.circular(4)
                              },
                              child: Text(`${item.value}%`, {
                                style: TextStyle({
                                  color: 'white',
                                  fontSize: 12,
                                  fontWeight: 'bold'
                                })
                              })
                            })
                          }),
                          // 바
                          AnimatedContainer({
                            duration: 300,
                            width: barWidth,
                            height: height,
                            decoration: {
                              color: item.color,
                              borderRadius: BorderRadius.only({
                                topLeft: 8,
                                topRight: 8
                              }),
                              opacity: isSelected ? 1 : (isHovered ? 0.8 : 0.6)
                            }
                          }),
                          SizedBox({ height: 12 }),
                          Text(item.label, {
                            style: TextStyle({
                              fontSize: 14,
                              fontWeight: isSelected ? 'bold' : 'normal',
                              color: isSelected ? item.color : '#495057'
                            })
                          })
                        ]
                      })
                    });
                  })
                })
              ]
            })
          })
        ]
      })
    });
  })()}
</Widget>

## CustomPaint를 사용한 바 차트

더 정교한 컨트롤을 위해 CustomPaint를 사용합니다.

<Widget name="cookbook-custom-paint-bar-chart">
  {(() => {
    const [tooltip, setTooltip] = React.useState(null);
    
    const data = [
      { month: '1월', sales: 4200, profit: 2100 },
      { month: '2월', sales: 3800, profit: 1900 },
      { month: '3월', sales: 5100, profit: 2800 },
      { month: '4월', sales: 4600, profit: 2400 },
      { month: '5월', sales: 5400, profit: 3200 },
      { month: '6월', sales: 4900, profit: 2600 }
    ];
    
    const maxValue = Math.max(...data.flatMap(d => [d.sales, d.profit]));
    
    const BarChartPainter = {
      paint: (canvas, size) => {
        const paint = Paint();
        const textPaint = Paint();
        
        const padding = 40;
        const chartWidth = size.width - padding * 2;
        const chartHeight = size.height - padding * 2;
        const barWidth = chartWidth / (data.length * 2 + 1);
        
        // 배경 그리드
        paint.color = '#e9ecef';
        paint.strokeWidth = 1;
        
        for (let i = 0; i <= 5; i++) {
          const y = padding + (chartHeight / 5) * i;
          canvas.drawLine(
            { x: padding, y },
            { x: size.width - padding, y },
            paint
          );
        }
        
        // 바 그리기
        data.forEach((item, index) => {
          const x = padding + (index * 2 + 0.5) * barWidth;
          
          // 매출 바
          const salesHeight = (item.sales / maxValue) * chartHeight;
          paint.color = '#4c6ef5';
          canvas.drawRRect(
            RRect.fromRectAndCorners(
              Rect.fromLTWH(x, size.height - padding - salesHeight, barWidth * 0.8, salesHeight),
              topLeft: Radius.circular(4),
              topRight: Radius.circular(4)
            ),
            paint
          );
          
          // 수익 바
          const profitHeight = (item.profit / maxValue) * chartHeight;
          paint.color = '#51cf66';
          canvas.drawRRect(
            RRect.fromRectAndCorners(
              Rect.fromLTWH(x + barWidth, size.height - padding - profitHeight, barWidth * 0.8, profitHeight),
              topLeft: Radius.circular(4),
              topRight: Radius.circular(4)
            ),
            paint
          );
          
          // 월 레이블
          textPaint.color = '#495057';
          textPaint.fontSize = 12;
          canvas.drawText(
            item.month,
            { x: x + barWidth / 2, y: size.height - 20 },
            textPaint
          );
        });
        
        // Y축 레이블
        for (let i = 0; i <= 5; i++) {
          const value = Math.round((maxValue / 5) * (5 - i));
          const y = padding + (chartHeight / 5) * i;
          
          textPaint.color = '#868e96';
          textPaint.fontSize = 10;
          canvas.drawText(
            value.toString(),
            { x: 10, y: y },
            textPaint
          );
        }
      }
    };
    
    return Container({
      width: 450,
      height: 350,
      color: 'white',
      padding: EdgeInsets.all(20),
      child: Column({
        children: [
          Row({
            mainAxisAlignment: 'spaceBetween',
            children: [
              Text('월별 매출/수익', {
                style: TextStyle({
                  fontSize: 18,
                  fontWeight: 'bold',
                  color: '#212529'
                })
              }),
              Row({
                children: [
                  Row({
                    children: [
                      Container({
                        width: 12,
                        height: 12,
                        color: '#4c6ef5'
                      }),
                      SizedBox({ width: 4 }),
                      Text('매출', {
                        style: TextStyle({
                          fontSize: 12,
                          color: '#495057'
                        })
                      })
                    ]
                  }),
                  SizedBox({ width: 16 }),
                  Row({
                    children: [
                      Container({
                        width: 12,
                        height: 12,
                        color: '#51cf66'
                      }),
                      SizedBox({ width: 4 }),
                      Text('수익', {
                        style: TextStyle({
                          fontSize: 12,
                          color: '#495057'
                        })
                      })
                    ]
                  })
                ]
              })
            ]
          }),
          SizedBox({ height: 20 }),
          Expanded({
            child: CustomPaint({
              painter: BarChartPainter
            })
          })
        ]
      })
    });
  })()}
</Widget>

## 구현 팁

### 1. 데이터 정규화
```typescript
const maxValue = Math.max(...data.map(d => d.value));
const normalizedHeight = (value / maxValue) * chartHeight;
```

### 2. 반응형 디자인
```typescript
// 컨테이너 크기에 따라 바 너비 조정
const barWidth = containerWidth / (data.length * 2);
```

### 3. 애니메이션 효과
```typescript
AnimatedContainer({
  duration: 300,
  height: barHeight,
  curve: Curves.easeInOut
})
```

### 4. 접근성 개선
- 각 바에 aria-label 추가
- 키보드 네비게이션 지원
- 색맹 사용자를 위한 패턴 추가

## 성능 최적화

1. **대량 데이터**: 가상화 또는 집계 사용
2. **애니메이션**: requestAnimationFrame 활용
3. **메모리**: 불필요한 리렌더링 방지

## 확장 아이디어

1. **축 레이블**: X/Y축에 상세 레이블 추가
2. **그리드 라인**: 값 읽기 쉽게 배경 그리드
3. **데이터 레이블**: 바 위에 정확한 값 표시
4. **범례**: 여러 시리즈 구분을 위한 범례
5. **스크롤**: 많은 데이터를 위한 수평 스크롤
