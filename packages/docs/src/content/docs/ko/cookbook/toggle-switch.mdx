import { Widget } from '@meursyphus/flitter-svelte';
import { Container, Row, Column, Text, TextStyle, GestureDetector, AnimatedContainer, AnimatedAlign, Center, EdgeInsets, BorderRadius, BoxShadow, Stack } from '@meursyphus/flitter';

# 토글 스위치

토글 스위치는 설정이나 옵션을 켜고 끄는 데 사용되는 일반적인 UI 컴포넌트입니다. 애니메이션을 활용하여 부드러운 전환 효과를 구현할 수 있습니다.

## 기본 토글 스위치

가장 간단한 형태의 토글 스위치입니다.

<Widget name="cookbook-toggle-basic">
  {(() => {
    const [isOn, setIsOn] = React.useState(false);
    
    return Container({
      width: 300,
      height: 150,
      color: '#f8f9fa',
      child: Center({
        child: GestureDetector({
          onTap: () => setIsOn(!isOn),
          child: AnimatedContainer({
            duration: 200,
            width: 60,
            height: 30,
            decoration: {
              color: isOn ? '#51cf66' : '#e9ecef',
              borderRadius: BorderRadius.circular(15)
            },
            child: AnimatedAlign({
              duration: 200,
              alignment: isOn ? Alignment.centerRight : Alignment.centerLeft,
              child: Container({
                width: 26,
                height: 26,
                margin: EdgeInsets.all(2),
                decoration: {
                  color: 'white',
                  shape: 'circle',
                  boxShadow: [
                    BoxShadow({
                      color: 'rgba(0,0,0,0.2)',
                      blurRadius: 4,
                      offset: { x: 0, y: 2 }
                    })
                  ]
                }
              })
            })
          })
        })
      })
    });
  })()}
</Widget>


## 라벨이 있는 토글 스위치

설명 텍스트와 함께 표시되는 토글 스위치입니다.

<Widget name="cookbook-toggle-labeled">
  {(() => {
    const [settings, setSettings] = React.useState({
      notifications: true,
      darkMode: false,
      autoSave: true
    });
    
    const ToggleItem = ({ label, value, onChange }) => (
      Container({
        padding: EdgeInsets.symmetric({ horizontal: 20, vertical: 12 }),
        margin: EdgeInsets.only({ bottom: 8 }),
        decoration: {
          color: 'white',
          borderRadius: BorderRadius.circular(12),
          boxShadow: [
            BoxShadow({
              color: 'rgba(0,0,0,0.05)',
              blurRadius: 4,
              offset: { x: 0, y: 2 }
            })
          ]
        },
        child: Row({
          mainAxisAlignment: 'spaceBetween',
          children: [
            Text(label, {
              style: TextStyle({
                fontSize: 16,
                color: '#495057'
              })
            }),
            GestureDetector({
              onTap: onChange,
              child: AnimatedContainer({
                duration: 200,
                width: 50,
                height: 28,
                decoration: {
                  color: value ? '#4c6ef5' : '#dee2e6',
                  borderRadius: BorderRadius.circular(14)
                },
                child: AnimatedAlign({
                  duration: 200,
                  alignment: value ? Alignment.centerRight : Alignment.centerLeft,
                  child: Container({
                    width: 24,
                    height: 24,
                    margin: EdgeInsets.all(2),
                    decoration: {
                      color: 'white',
                      shape: 'circle'
                    }
                  })
                })
              })
            })
          ]
        })
      })
    );
    
    return Container({
      width: 350,
      padding: EdgeInsets.all(20),
      color: '#f8f9fa',
      child: Column({
        crossAxisAlignment: 'start',
        children: [
          Text('설정', {
            style: TextStyle({
              fontSize: 24,
              fontWeight: 'bold',
              color: '#212529'
            })
          }),
          SizedBox({ height: 20 }),
          ToggleItem({
            label: '알림 받기',
            value: settings.notifications,
            onChange: () => setSettings({
              ...settings,
              notifications: !settings.notifications
            })
          }),
          ToggleItem({
            label: '다크 모드',
            value: settings.darkMode,
            onChange: () => setSettings({
              ...settings,
              darkMode: !settings.darkMode
            })
          }),
          ToggleItem({
            label: '자동 저장',
            value: settings.autoSave,
            onChange: () => setSettings({
              ...settings,
              autoSave: !settings.autoSave
            })
          })
        ]
      })
    });
  })()}
</Widget>

## iOS 스타일 토글 스위치

iOS 스타일의 토글 스위치 구현입니다.

<Widget name="cookbook-toggle-ios">
  {(() => {
    const [isOn, setIsOn] = React.useState(false);
    
    return Container({
      width: 350,
      height: 200,
      color: '#f8f9fa',
      child: Center({
        child: Column({
          mainAxisSize: 'min',
          children: [
            GestureDetector({
              onTap: () => setIsOn(!isOn),
              child: Container({
                width: 80,
                height: 44,
                child: Stack({
                  children: [
                    AnimatedContainer({
                      duration: 200,
                      decoration: {
                        color: isOn ? '#34c759' : '#e9e9ea',
                        borderRadius: BorderRadius.circular(22)
                      }
                    }),
                    AnimatedPositioned({
                      duration: 200,
                      left: isOn ? 38 : 2,
                      top: 2,
                      child: Container({
                        width: 40,
                        height: 40,
                        decoration: {
                          color: 'white',
                          shape: 'circle',
                          boxShadow: [
                            BoxShadow({
                              color: 'rgba(0,0,0,0.2)',
                              blurRadius: 6,
                              offset: { x: 0, y: 3 }
                            })
                          ]
                        }
                      })
                    })
                  ]
                })
              })
            }),
            SizedBox({ height: 16 }),
            Text(isOn ? '켜짐' : '꺼짐', {
              style: TextStyle({
                fontSize: 16,
                color: isOn ? '#34c759' : '#8e8e93'
              })
            })
          ]
        })
      })
    });
  })()}
</Widget>

## 커스텀 디자인 토글 스위치

아이콘과 색상이 변경되는 커스텀 토글 스위치입니다.

<Widget name="cookbook-toggle-custom">
  {(() => {
    const [isDarkMode, setIsDarkMode] = React.useState(false);
    
    return Container({
      width: 350,
      height: 250,
      decoration: {
        color: isDarkMode ? '#1a1b1e' : '#ffffff',
        borderRadius: BorderRadius.circular(20)
      },
      child: Center({
        child: Column({
          mainAxisSize: 'min',
          children: [
            Text('테마 설정', {
              style: TextStyle({
                fontSize: 20,
                fontWeight: 'bold',
                color: isDarkMode ? 'white' : '#212529'
              })
            }),
            SizedBox({ height: 30 }),
            GestureDetector({
              onTap: () => setIsDarkMode(!isDarkMode),
              child: Container({
                width: 120,
                height: 60,
                padding: EdgeInsets.all(4),
                decoration: {
                  color: isDarkMode ? '#2c2d30' : '#f1f3f5',
                  borderRadius: BorderRadius.circular(30),
                  border: {
                    color: isDarkMode ? '#495057' : '#dee2e6',
                    width: 2
                  }
                },
                child: AnimatedAlign({
                  duration: 300,
                  alignment: isDarkMode ? Alignment.centerRight : Alignment.centerLeft,
                  curve: Curves.easeInOut,
                  child: Container({
                    width: 52,
                    height: 52,
                    decoration: {
                      gradient: {
                        type: 'linear',
                        colors: isDarkMode 
                          ? ['#4c6ef5', '#5c7cfa'] 
                          : ['#ffd43b', '#fab005'],
                        begin: 'topLeft',
                        end: 'bottomRight'
                      },
                      shape: 'circle',
                      boxShadow: [
                        BoxShadow({
                          color: isDarkMode 
                            ? 'rgba(76,110,245,0.3)' 
                            : 'rgba(255,212,59,0.3)',
                          blurRadius: 8,
                          offset: { x: 0, y: 4 }
                        })
                      ]
                    },
                    child: Center({
                      child: Text(isDarkMode ? '🌙' : '☀️', {
                        style: TextStyle({ fontSize: 24 })
                      })
                    })
                  })
                })
              })
            }),
            SizedBox({ height: 20 }),
            Text(isDarkMode ? '다크 모드' : '라이트 모드', {
              style: TextStyle({
                fontSize: 16,
                color: isDarkMode ? '#adb5bd' : '#495057'
              })
            })
          ]
        })
      })
    });
  })()}
</Widget>

## 상태 표시가 있는 토글 스위치

진행 상태를 보여주는 토글 스위치입니다.

<Widget name="cookbook-toggle-loading">
  {(() => {
    const [isOn, setIsOn] = React.useState(false);
    const [isLoading, setIsLoading] = React.useState(false);
    
    const handleToggle = async () => {
      setIsLoading(true);
      // 비동기 작업 시뮬레이션
      await new Promise(resolve => setTimeout(resolve, 1000));
      setIsOn(!isOn);
      setIsLoading(false);
    };
    
    return Container({
      width: 350,
      height: 200,
      color: '#f8f9fa',
      child: Center({
        child: Column({
          mainAxisSize: 'min',
          children: [
            Text('Wi-Fi', {
              style: TextStyle({
                fontSize: 18,
                fontWeight: 'bold',
                color: '#212529'
              })
            }),
            SizedBox({ height: 20 }),
            GestureDetector({
              onTap: isLoading ? null : handleToggle,
              child: Container({
                width: 70,
                height: 35,
                decoration: {
                  color: isOn ? '#40c057' : '#e9ecef',
                  borderRadius: BorderRadius.circular(17.5)
                },
                child: Stack({
                  children: [
                    AnimatedAlign({
                      duration: 200,
                      alignment: isOn ? Alignment.centerRight : Alignment.centerLeft,
                      child: Container({
                        width: 31,
                        height: 31,
                        margin: EdgeInsets.all(2),
                        decoration: {
                          color: 'white',
                          shape: 'circle',
                          boxShadow: [
                            BoxShadow({
                              color: 'rgba(0,0,0,0.2)',
                              blurRadius: 4,
                              offset: { x: 0, y: 2 }
                            })
                          ]
                        },
                        child: isLoading 
                          ? Center({
                              child: Container({
                                width: 20,
                                height: 20,
                                child: CircularProgressIndicator({
                                  strokeWidth: 2,
                                  color: '#40c057'
                                })
                              })
                            })
                          : null
                      })
                    })
                  ]
                })
              })
            }),
            SizedBox({ height: 16 }),
            Text(
              isLoading ? '연결 중...' : (isOn ? '연결됨' : '연결 끊김'),
              {
                style: TextStyle({
                  fontSize: 14,
                  color: isLoading ? '#fab005' : (isOn ? '#40c057' : '#868e96')
                })
              }
            )
          ]
        })
      })
    });
  })()}
</Widget>

## 구현 팁

### 1. 애니메이션 타이밍
```typescript
// 동일한 duration 사용으로 동기화
AnimatedContainer({ duration: 200 })
AnimatedAlign({ duration: 200 })
```

### 2. 터치 영역 확대
```typescript
GestureDetector({
  behavior: HitTestBehavior.opaque,
  onTap: handleToggle,
  child: Padding({
    padding: EdgeInsets.all(8), // 터치 영역 확대
    child: Switch()
  })
})
```

### 3. 접근성 개선
```typescript
Semantics({
  label: '알림 설정',
  value: isOn ? '켜짐' : '꺼짐',
  child: ToggleSwitch()
})
```

## 커스터마이징 옵션

1. **크기**: width/height 조정으로 다양한 크기 구현
2. **색상**: 브랜드 컬러에 맞게 색상 변경
3. **모양**: borderRadius 조정으로 둥근 정도 변경
4. **애니메이션**: duration과 curve로 움직임 커스터마이징
