import { Widget } from '@meursyphus/flitter-svelte';
import { Container, Column, Text, TextStyle, GestureDetector, Center, AnimatedOpacity, EdgeInsets, BorderRadius, AnimationController, Curves } from '@meursyphus/flitter';

# 페이드 인/아웃 효과

페이드 효과는 UI 요소를 부드럽게 나타나고 사라지게 만드는 가장 기본적인 애니메이션입니다. Flitter의 `AnimatedOpacity` 위젯을 사용하여 쉽게 구현할 수 있습니다.

## 기본 페이드 효과

버튼을 클릭하여 요소를 페이드 인/아웃시키는 기본 예제입니다.

<Widget name="cookbook-fade-basic">
  {(() => {
    const [isVisible, setIsVisible] = React.useState(true);
    
    return Container({
      width: 300,
      height: 250,
      color: '#f8f9fa',
      child: Column({
        mainAxisAlignment: 'center',
        children: [
          AnimatedOpacity({
            opacity: isVisible ? 1.0 : 0.0,
            duration: 500, // 밀리초
            child: Container({
              width: 150,
              height: 150,
              decoration: {
                color: '#4c6ef5',
                borderRadius: BorderRadius.circular(16)
              },
              child: Center({
                child: Text('안녕하세요!', {
                  style: TextStyle({
                    color: 'white',
                    fontSize: 20,
                    fontWeight: 'bold'
                  })
                })
              })
            })
          }),
          SizedBox({ height: 20 }),
          GestureDetector({
            onTap: () => setIsVisible(!isVisible),
            child: Container({
              padding: EdgeInsets.symmetric({ horizontal: 24, vertical: 12 }),
              decoration: {
                color: '#495057',
                borderRadius: BorderRadius.circular(8)
              },
              child: Text(isVisible ? '숨기기' : '보이기', {
                style: TextStyle({
                  color: 'white',
                  fontWeight: 'bold'
                })
              })
            })
          })
        ]
      })
    });
  })()}
</Widget>

```typescript
AnimatedOpacity({
  opacity: isVisible ? 1.0 : 0.0,
  duration: 500, // 밀리초
  child: YourWidget()
})
```

## 순차적 페이드 효과

여러 요소가 순차적으로 나타나는 효과입니다.

<Widget name="cookbook-fade-staggered">
  {(() => {
    const [showItems, setShowItems] = React.useState(false);
    
    const items = ['첫 번째', '두 번째', '세 번째', '네 번째'];
    
    return Container({
      width: 350,
      height: 400,
      color: '#f8f9fa',
      padding: EdgeInsets.all(20),
      child: Column({
        children: [
          Text('순차적 페이드 효과', {
            style: TextStyle({
              fontSize: 24,
              fontWeight: 'bold',
              color: '#212529'
            })
          }),
          SizedBox({ height: 20 }),
          ...items.map((item, index) => 
            AnimatedOpacity({
              opacity: showItems ? 1.0 : 0.0,
              duration: 300 + (index * 200), // 각 아이템마다 지연
              child: Container({
                width: double.infinity,
                margin: EdgeInsets.only({ bottom: 10 }),
                padding: EdgeInsets.all(16),
                decoration: {
                  color: 'white',
                  borderRadius: BorderRadius.circular(8),
                  boxShadow: [{
                    color: 'rgba(0,0,0,0.1)',
                    blurRadius: 4,
                    offset: { x: 0, y: 2 }
                  }]
                },
                child: Text(item, {
                  style: TextStyle({
                    fontSize: 16,
                    color: '#495057'
                  })
                })
              })
            })
          ),
          SizedBox({ height: 20 }),
          GestureDetector({
            onTap: () => setShowItems(!showItems),
            child: Container({
              padding: EdgeInsets.symmetric({ horizontal: 32, vertical: 12 }),
              decoration: {
                color: '#5c7cfa',
                borderRadius: BorderRadius.circular(24)
              },
              child: Text(showItems ? '숨기기' : '보이기', {
                style: TextStyle({
                  color: 'white',
                  fontWeight: 'bold'
                })
              })
            })
          })
        ]
      })
    });
  })()}
</Widget>

## 페이드와 함께 크기 변경

페이드 효과와 함께 크기도 변경되는 애니메이션입니다.

<Widget name="cookbook-fade-scale">
  {(() => {
    const [isExpanded, setIsExpanded] = React.useState(false);
    
    return Container({
      width: 350,
      height: 300,
      color: '#e9ecef',
      child: Center({
        child: Column({
          mainAxisSize: 'min',
          children: [
            GestureDetector({
              onTap: () => setIsExpanded(!isExpanded),
              child: Container({
                padding: EdgeInsets.all(16),
                decoration: {
                  color: '#ff6b6b',
                  borderRadius: BorderRadius.circular(12)
                },
                child: Row({
                  mainAxisSize: 'min',
                  children: [
                    Text('메뉴', {
                      style: TextStyle({
                        color: 'white',
                        fontSize: 18,
                        fontWeight: 'bold'
                      })
                    }),
                    SizedBox({ width: 8 }),
                    AnimatedRotation({
                      turns: isExpanded ? 0.5 : 0,
                      duration: 300,
                      child: Text('▼', {
                        style: TextStyle({
                          color: 'white',
                          fontSize: 14
                        })
                      })
                    })
                  ]
                })
              })
            }),
            AnimatedOpacity({
              opacity: isExpanded ? 1.0 : 0.0,
              duration: 300,
              child: AnimatedContainer({
                duration: 300,
                height: isExpanded ? 150 : 0,
                margin: EdgeInsets.only({ top: isExpanded ? 10 : 0 }),
                child: Container({
                  width: 200,
                  padding: EdgeInsets.all(16),
                  decoration: {
                    color: 'white',
                    borderRadius: BorderRadius.circular(12),
                    boxShadow: [{
                      color: 'rgba(0,0,0,0.1)',
                      blurRadius: 8,
                      offset: { x: 0, y: 4 }
                    }]
                  },
                  child: Column({
                    crossAxisAlignment: 'start',
                    children: [
                      Text('• 홈으로', { style: TextStyle({ fontSize: 16 }) }),
                      SizedBox({ height: 8 }),
                      Text('• 프로필', { style: TextStyle({ fontSize: 16 }) }),
                      SizedBox({ height: 8 }),
                      Text('• 설정', { style: TextStyle({ fontSize: 16 }) }),
                      SizedBox({ height: 8 }),
                      Text('• 로그아웃', { style: TextStyle({ fontSize: 16 }) })
                    ]
                  })
                })
              })
            })
          ]
        })
      })
    });
  })()}
</Widget>

              children: [
                AnimatedOpacity({

## 고급 기법

### 1. 커브 사용하기
```typescript
AnimatedOpacity({
  opacity: visible ? 1.0 : 0.0,
  duration: 500,
  curve: Curves.easeInOut, // 다양한 커브 옵션
  child: YourWidget()
})
```

### 2. 애니메이션 콜백
```typescript
AnimatedOpacity({
  opacity: visible ? 1.0 : 0.0,
  duration: 500,
  onEnd: () => {
    console.log('애니메이션 완료!');
  },
  child: YourWidget()
})
```

### 3. 조건부 렌더링과 함께 사용
```typescript
Stack({
  children: [
    if (showA) AnimatedOpacity({
      opacity: 1.0,
      duration: 300,
      child: WidgetA()
    }),
    if (showB) AnimatedOpacity({
      opacity: 1.0,
      duration: 300,
      child: WidgetB()
    })
  ]
})
```

## 성능 팁

1. **duration 최적화**: 너무 긴 애니메이션은 사용자 경험을 해칩니다. 일반적으로 200-500ms가 적당합니다.

2. **여러 애니메이션 동시 실행**: 많은 요소를 동시에 애니메이션하면 성능이 저하될 수 있습니다.

3. **opacity: 0 vs 제거**: opacity가 0이어도 위젯은 여전히 공간을 차지하고 렌더링됩니다. 완전히 제거하려면 조건부 렌더링을 사용하세요.
