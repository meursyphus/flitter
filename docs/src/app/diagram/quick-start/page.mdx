# Quick Start

<QuickStartDemo />

1. Run the three commands from [Installation](/diagram/installation).
2. Save this file as `app.tsx`.
3. Drag, connect, press ⌘Z, right-click.

```tsx
import Widget from "@flitterjs/react";
import { layeredLayout, type FlowEdge, type FlowNode } from "flitter-ui/diagram";
import { FlowDiagram } from "@/components/diagram/flow-diagram";

// Positions are placeholders; the layout places every node.
const nodes: FlowNode[] = [
  { id: "draft", type: "input", position: { x: 0, y: 0 }, data: { label: "Draft" } },
  { id: "review", position: { x: 0, y: 0 }, data: { label: "Review" } },
  { id: "publish", type: "output", position: { x: 0, y: 0 }, data: { label: "Publish" } },
];

const edges: FlowEdge[] = [
  { id: "a", source: "draft", target: "review" },
  { id: "b", source: "review", target: "publish", markerEnd: "arrowclosed" },
];

// Create the widget once. A new instance remounts the editor.
const diagram = FlowDiagram({
  nodes,
  edges,
  layout: { algorithm: layeredLayout, options: { direction: "TB", rankSpacing: 80 } },
  viewport: { fitView: { padding: 0.3 } },
  interaction: { drag: { snap: { grid: [20, 20] } } },
  on: {
    connect: (connection) => console.log("connected", connection),
    nodeClick: (_event, node) => console.log("clicked", node.id),
    historyChange: ({ canUndo }) => console.log("can undo", canUndo),
  },
});

export default function App() {
  return <Widget widget={diagram} width="100%" height="480px" renderer="svg" />;
}
```
